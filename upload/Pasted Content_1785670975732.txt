    <!-- JavaScript Application Logic -->
    <script>
        // Initial Default Data setup
        const initialBranches = [
            { id: 'b1', name: 'بن غرسه' },
            { id: 'b2', name: 'ابونواس' },
            { id: 'b3', name: 'المنصور' },
            { id: 'b4', name: 'عين زاره' }
        ];

        const initialEmployees = [
            { id: 'e1', branchId: 'b1', name: 'أحمد', shift: 'الفترة الصباحية' },
            { id: 'e2', branchId: 'b1', name: 'هاشم', shift: 'الفترة المسائية' },
            { id: 'e3', branchId: 'b2', name: 'هيثم', shift: 'الفترة الصباحية' },
            { id: 'e4', branchId: 'b2', name: 'اياد', shift: 'الفترة المسائية' },
            { id: 'e5', branchId: 'b3', name: 'وسام', shift: 'الفترة كاملة' },
            { id: 'e6', branchId: 'b4', name: 'اسامه', shift: 'الفترة الصباحية' },
            { id: 'e7', branchId: 'b4', name: 'اياد', shift: 'الفترة المسائية' }
        ];

        // Application State
        let branches = JSON.parse(localStorage.getItem('gc_branches')) || initialBranches;
        let employees = JSON.parse(localStorage.getItem('gc_employees')) || initialEmployees;
        let records = JSON.parse(localStorage.getItem('gc_records')) || [];
        let closedDays = JSON.parse(localStorage.getItem('gc_closedDays')) || {}; // { 'YYYY-MM-DD': true }

        // App Init
        window.onload = () => {
            document.getElementById('datePicker').value = todayISO();
            renderApp();
        };

        function saveData() {
            localStorage.setItem('gc_branches', JSON.stringify(branches));
            localStorage.setItem('gc_employees', JSON.stringify(employees));
            localStorage.setItem('gc_records', JSON.stringify(records));
            localStorage.setItem('gc_closedDays', JSON.stringify(closedDays));
        }

        function isDayClosed(date) {
            return !!closedDays[date];
        }

        function renderApp() {
            const selectedDate = document.getElementById('datePicker').value;
            const dayClosed = isDayClosed(selectedDate);
            const container = document.getElementById('branchesContainer');
            container.innerHTML = '';

            const banner = document.getElementById('dayStatusBanner');
            banner.innerHTML = dayClosed
                ? `<div class="bg-violet-500/10 border border-violet-500/30 text-violet-300 rounded-xl p-4 flex justify-between items-center text-sm">
                        <span>🔒 هذا اليوم (${formatDateShort(selectedDate)}) <strong>مغلق</strong> — لا يمكن إضافة أو تعديل أو حذف الحركات المالية له.</span>
                        <button onclick="openClosingModal()" class="text-violet-300 hover:text-white underline text-xs">عرض تفاصيل الإغلاق</button>
                   </div>`
                : '';

            let grandWithdrawals = 0;
            let grandShortages = 0;

            branches.forEach(branch => {
                const branchEmps = employees.filter(e => e.branchId === branch.id);
                
                let branchHtml = `
                    <div class="bg-slate-800 border border-slate-700 rounded-2xl p-6 shadow-lg">
                        <div class="flex justify-between items-center mb-6 pb-4 border-b border-slate-700">
                            <h2 class="text-xl font-bold text-cyan-400 flex items-center gap-2">📍 ${branch.name}</h2>
                            <button onclick="deleteBranch('${branch.id}')" class="text-rose-400 hover:text-rose-300 text-xs font-semibold bg-rose-500/10 px-3 py-1.5 rounded-lg border border-rose-500/20">حذف الفرع</button>
                        </div>
                        <div class="space-y-4">
                `;

                if (branchEmps.length === 0) {
                    branchHtml += `<p class="text-slate-500 text-sm text-center py-4">لا يوجد موظفون بالفرع حالياً</p>`;
                }

                branchEmps.forEach(emp => {
                    const empRecords = records.filter(r => r.empId === emp.id && r.date === selectedDate).sort((a, b) => (b.date || '').localeCompare(a.date || ''));
                    const withdrawals = empRecords.filter(r => r.type === 'withdrawal').reduce((sum, r) => sum + r.amount, 0);
                    const shortages = empRecords.filter(r => r.type === 'shortage').reduce((sum, r) => sum + r.amount, 0);

                    grandWithdrawals += withdrawals;
                    grandShortages += shortages;

                    branchHtml += `
                        <div class="bg-slate-900/70 border border-slate-700/60 rounded-xl p-4">
                            <div class="flex justify-between items-start mb-3">
                                <div>
                                    <h3 class="font-bold text-white text-base">${emp.name}</h3>
                                    <span class="text-xs text-indigo-400 bg-indigo-500/10 px-2.5 py-0.5 rounded-full mt-1 inline-block border border-indigo-500/20">${emp.shift}</span>
                                </div>
                                <div class="flex gap-2">
                                    ${dayClosed
                                        ? `<button disabled class="bg-slate-700 text-slate-500 text-xs px-2.5 py-1.5 rounded-lg border border-slate-600 cursor-not-allowed">🔒 مغلق</button>`
                                        : `<button onclick="openRecordModal('${emp.id}', '${emp.name}')" class="bg-cyan-600/20 hover:bg-cyan-600 text-cyan-400 hover:text-white text-xs px-2.5 py-1.5 rounded-lg border border-cyan-500/30 transition">+ حركة</button>`
                                    }
                                    <button onclick="deleteEmployee('${emp.id}')" class="text-slate-500 hover:text-rose-400 text-xs p-1">🗑️</button>
                                </div>
                            </div>
                            
                            <!-- Financial Totals -->
                            <div class="grid grid-cols-2 gap-3 bg-slate-800/80 p-2.5 rounded-lg text-xs mb-3">
                                <div class="flex justify-between">
                                    <span class="text-slate-400">السحبيات:</span>
                                    <span class="font-bold text-amber-400">${withdrawals} د.ل</span>
                                </div>
                                <div class="flex justify-between border-r border-slate-700 pr-3">
                                    <span class="text-slate-400">العجوزات:</span>
                                    <span class="font-bold text-rose-400">${shortages} د.ل</span>
                                </div>
                            </div>

                            <!-- Record History -->
                            <div class="space-y-1">
                                ${empRecords.map(r => `
                                    <div class="flex justify-between items-center text-xs text-slate-400 bg-slate-900 px-2.5 py-1 rounded">
                                        <span>${r.type === 'withdrawal' ? '💸 سحب' : '📉 عجز'} — ${formatDateShort(r.date)}: ${r.note || 'بدون ملاحظة'}</span>
                                        <div class="flex items-center gap-2">
                                            <span class="font-semibold ${r.type === 'withdrawal' ? 'text-amber-400' : 'text-rose-400'}">${r.amount} د.ل</span>
                                            ${dayClosed
                                                ? ''
                                                : `<button onclick="editRecord('${r.id}')" class="text-slate-600 hover:text-cyan-400">✏️</button>
                                                   <button onclick="deleteRecord('${r.id}')" class="text-slate-600 hover:text-rose-400">×</button>`
                                            }
                                        </div>
                                    </div>
                                `).join('')}
                            </div>
                        </div>
                    `;
                });

                branchHtml += `</div></div>`;
                container.innerHTML += branchHtml;
            });

            document.getElementById('totalWithdrawals').innerText = `${grandWithdrawals} د.ل`;
            document.getElementById('totalShortages').innerText = `${grandShortages} د.ل`;
        }

        // Modals Logic
        function todayISO() {
            const d = new Date();
            return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
        }

        function formatDateShort(dateStr) {
            if (!dateStr) return '';
            const [y, m, d] = dateStr.split('-');
            return `${d}/${m}/${y}`;
        }

        function openRecordModal(empId, empName) {
            if (isDayClosed(document.getElementById('datePicker').value)) {
                return alert('هذا اليوم مغلق. الرجاء إعادة فتحه من نافذة "الإغلاق اليومي" أولاً للتعديل.');
            }
            document.getElementById('recordId').value = '';
            document.getElementById('recordEmpId').value = empId;
            document.getElementById('modalTitle').innerText = `إضافة حركة لـ (${empName})`;
            document.getElementById('recordType').value = 'withdrawal';
            document.getElementById('recordDate').value = document.getElementById('datePicker').value || todayISO();
            document.getElementById('recordAmount').value = '';
            document.getElementById('recordNote').value = '';
            document.getElementById('recordModal').classList.remove('hidden');
            document.getElementById('recordModal').classList.add('flex');
        }

        function editRecord(recordId) {
            const r = records.find(rec => rec.id === recordId);
            if (!r) return;
            if (isDayClosed(r.date)) {
                return alert('هذا اليوم مغلق. الرجاء إعادة فتحه من نافذة "الإغلاق اليومي" أولاً للتعديل.');
            }
            const emp = employees.find(e => e.id === r.empId);
            document.getElementById('recordId').value = r.id;
            document.getElementById('recordEmpId').value = r.empId;
            document.getElementById('modalTitle').innerText = `تعديل حركة لـ (${emp ? emp.name : ''})`;
            document.getElementById('recordType').value = r.type;
            document.getElementById('recordDate').value = r.date || todayISO();
            document.getElementById('recordAmount').value = r.amount;
            document.getElementById('recordNote').value = r.note || '';
            document.getElementById('recordModal').classList.remove('hidden');
            document.getElementById('recordModal').classList.add('flex');
        }

        function openBranchModal() {
            document.getElementById('branchModal').classList.remove('hidden');
            document.getElementById('branchModal').classList.add('flex');
        }

        function openEmpModal() {
            const select = document.getElementById('empBranch');
            select.innerHTML = branches.map(b => `<option value="${b.id}">${b.name}</option>`).join('');
            document.getElementById('empModal').classList.remove('hidden');
            document.getElementById('empModal').classList.add('flex');
        }

        function closeModal(id) {
            document.getElementById(id).classList.remove('flex');
            document.getElementById(id).classList.add('hidden');
        }

        // Actions
        function saveRecord() {
            const recordId = document.getElementById('recordId').value;
            const empId = document.getElementById('recordEmpId').value;
            const type = document.getElementById('recordType').value;
            const amount = parseFloat(document.getElementById('recordAmount').value);
            const note = document.getElementById('recordNote').value;
            const date = document.getElementById('recordDate').value;

            if (!amount || amount <= 0) return alert('الرجاء إدخال مبلغ صحيح');
            if (!date) return alert('الرجاء تحديد تاريخ الحركة');
            if (isDayClosed(date)) return alert('لا يمكن حفظ حركة في يوم مغلق. الرجاء إعادة فتح اليوم أولاً.');

            const month = date.slice(0, 7); // YYYY-MM derived from the chosen date

            if (recordId) {
                // Editing an existing record
                const idx = records.findIndex(r => r.id === recordId);
                if (idx !== -1) {
                    records[idx] = { ...records[idx], type, amount, note, date, month };
                }
            } else {
                // Adding a new record
                records.push({ id: 'r_' + Date.now(), empId, type, amount, note, date, month });
            }

            saveData();
            closeModal('recordModal');
            document.getElementById('recordAmount').value = '';
            document.getElementById('recordNote').value = '';
            document.getElementById('recordId').value = '';

            // Switch the view to the day this record belongs to, so the user sees it immediately
            document.getElementById('datePicker').value = date;
            renderApp();
        }

        function deleteRecord(id) {
            const r = records.find(rec => rec.id === id);

        function formatMonthLabel(monthValue) {
            if (!monthValue) return '';
            const [year, month] = monthValue.split('-');
            return `${arabicMonths[parseInt(month, 10) - 1]} ${year}`;
        }

        function buildReportHTML(periodLabel, matchRecord) {

        function confirmCloseDay() {
            const selectedDate = document.getElementById('datePicker').value;
            if (!confirm(`هل أنت متأكد من إغلاق يوم ${formatDateShort(selectedDate)}؟ لن يمكن إضافة أو تعديل أو حذف حركات هذا اليوم إلا بعد إعادة فتحه.`)) return;
            closedDays[selectedDate] = true;
            saveData();
            renderClosingModalBody();
            renderApp();
        }

        function reopenDay() {
            const selectedDate = document.getElementById('datePicker').value;
            if (!confirm(`هل تريد إعادة فتح يوم ${formatDateShort(selectedDate)} للتعديل؟`)) return;
            delete closedDays[selectedDate];
            saveData();
            renderClosingModalBody();
            renderApp();
        }

        async function exportClosingPDF() {
            const selectedDate = document.getElementById('datePicker').value;
            const periodLabel = `إغلاق يومي: ${formatDateShort(selectedDate)}`;
            const matchRecord = r => r.date === selectedDate;

            try {
                const reportArea = document.getElementById('pdfReportArea');
                reportArea.innerHTML = buildReportHTML(periodLabel, matchRecord);
                await new Promise(resolve => setTimeout(resolve, 100));

                const canvas = await html2canvas(reportArea, { scale: 2, backgroundColor: '#ffffff', useCORS: true });
                const imgData = canvas.toDataURL('image/png');

                const { jsPDF } = window.jspdf;
                const pdf = new jsPDF('p', 'mm', 'a4');
                const pageWidth = pdf.internal.pageSize.getWidth();
                const pageHeight = pdf.internal.pageSize.getHeight();
                const imgWidth = pageWidth;
                const imgHeight = (canvas.height * imgWidth) / canvas.width;

                let heightLeft = imgHeight;
                let position = 0;
                pdf.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
                heightLeft -= pageHeight;
                while (heightLeft > 0) {
                    position = heightLeft - imgHeight;
                    pdf.addPage();
                    pdf.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
                    heightLeft -= pageHeight;
                }

                pdf.save(`إغلاق_يومي_${selectedDate}.pdf`);
                reportArea.innerHTML = '';
            } catch (err) {
                console.error(err);
                alert('حدث خطأ أثناء إنشاء ملف PDF. الرجاء المحاولة مرة أخرى.');
            }
        }

        async function exportPDF() {
            const type = document.getElementById('exportRangeType').value;

            let periodLabel, matchRecord, fileSuffix;

            if (type === 'month') {
                const monthValue = document.getElementById('exportMonth').value;
                if (!monthValue) return alert('الرجاء اختيار الشهر');
                periodLabel = `الشهر: ${formatMonthLabel(monthValue)}`;
                matchRecord = r => r.month === monthValue;
                fileSuffix = formatMonthLabel(monthValue).replace(/\s+/g, '_');
            } else if (type === 'day') {
                const dayValue = document.getElementById('exportDay').value;
                if (!dayValue) return alert('الرجاء اختيار اليوم');
                periodLabel = `اليوم: ${formatDateShort(dayValue)}`;
                matchRecord = r => r.date === dayValue;
                fileSuffix = dayValue;
            } else {
                const from = document.getElementById('exportFrom').value;
                const to = document.getElementById('exportTo').value;
                if (!from || !to) return alert('الرجاء تحديد بداية ونهاية الفترة');
                if (from > to) return alert('تاريخ البداية يجب أن يكون قبل تاريخ النهاية');
                periodLabel = `من ${formatDateShort(from)} إلى ${formatDateShort(to)}`;
                matchRecord = r => r.date >= from && r.date <= to;
                fileSuffix = `${from}_${to}`;
            }

            const confirmBtn = document.getElementById('exportConfirmBtn');
            const originalText = confirmBtn.innerHTML;
            confirmBtn.innerHTML = '⏳ جارٍ التصدير...';
            confirmBtn.disabled = true;

            try {
                const reportArea = document.getElementById('pdfReportArea');
                reportArea.innerHTML = buildReportHTML(periodLabel, matchRecord);

                // Give the browser a moment to layout the injected content
                await new Promise(resolve => setTimeout(resolve, 100));

                const canvas = await html2canvas(reportArea, { scale: 2, backgroundColor: '#ffffff', useCORS: true });
                const imgData = canvas.toDataURL('image/png');

                const { jsPDF } = window.jspdf;
                const pdf = new jsPDF('p', 'mm', 'a4');
                const pageWidth = pdf.internal.pageSize.getWidth();
                const pageHeight = pdf.internal.pageSize.getHeight();
                const imgWidth = pageWidth;
                const imgHeight = (canvas.height * imgWidth) / canvas.width;

                let heightLeft = imgHeight;
                let position = 0;

                pdf.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
                heightLeft -= pageHeight;

                while (heightLeft > 0) {
                    position = heightLeft - imgHeight;
                    pdf.addPage();
                    pdf.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
                    heightLeft -= pageHeight;
                }

                pdf.save(`تقرير_جيت_كلين_${fileSuffix}.pdf`);

                reportArea.innerHTML = '';
                closeModal('exportModal');
            } catch (err) {
                console.error(err);
                alert('حدث خطأ أثناء إنشاء ملف PDF. الرجاء المحاولة مرة أخرى.');
            } finally {
                confirmBtn.innerHTML = originalText;
                confirmBtn.disabled = false;
            }
        }
