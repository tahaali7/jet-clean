   119→function getRoomsForBranch(branchName: string) { return BRANCH_ROOMS[branchName] || ROOMS }
   120→function getPricesForRoom(room: string) { return ROOM_PRICES[room] || ALL_PRICES }
   121→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   122→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   123→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   124→  return Math.floor(totalAmount / 2) - deduction
   125→}
   126→function getNetFormulaText(branchName: string, roomName: string) {
   127→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return 'الإجمالي ÷ 2'
   128→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   129→  if (deduction === 0) return 'الإجمالي ÷ 2'
   130→  return `الإجمالي ÷ 2 - ${deduction}`
   131→}
   132→function todayISO() {
   133→  const d = new Date()
   134→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   135→}
   136→function formatDateShort(dateStr: string) {
   137→  if (!dateStr) return ''
   138→  const [y, m, d] = dateStr.split('-')
   139→  return `${d}/${m}/${y}`
   140→}

   147→  let roomTotalCars = 0

   150→  const mergedCounts: Record<string, number> = {}
   151→  const mergedCustoms: Record<string, { price: number; count: number }> = {}
   152→
   153→  roomEntries.forEach(entry => {
   154→    prices.forEach(price => {
   155→      const count = (entry.priceCounts && entry.priceCounts[price]) || 0
   156→      mergedCounts[price] = (mergedCounts[price] || 0) + count
   157→    })
   158→    if (entry.customPrices) {
   159→      Object.keys(entry.customPrices).forEach(key => {
   160→        const item = entry.customPrices[key]
   161→        if (mergedCustoms[key]) {
   162→          mergedCustoms[key].count += item.count
   163→        } else {
   164→          mergedCustoms[key] = { price: item.price, count: item.count }
   165→        }
   166→      })
   167→    }
   168→    roomTotalAmount += entry.totalAmount
   169→    roomTotalCars += entry.totalCars
   170→    roomExtraCars += (entry.extraCars || 0)
   171→    roomExtraAmount += (entry.extraAmount || 0)
   172→  })

   174→  const roomNet = getNetAmount(roomTotalAmount, branchName, room)
   175→  // Auto-adaptive: level 0=normal, 1=compact, 2=ultra-compact
   176→  const sl = sizeLevel || 0

   310→function buildCarReportHTML(selectedDate: string, branchId: string, branchName: string, entries: CarEntry[], savedWorkerExpenses?: Record<string, { cleanliness?: number; treasury?: Record<string, { income: number; expense: number }> }>): string[] {
   311→  // Group entries by room
   312→  const roomMap: Record<string, CarEntry[]> = {}
   313→  entries.forEach(e => {
   314→    if (!roomMap[e.room]) roomMap[e.room] = []
   315→    roomMap[e.room].push(e)
   316→  })
   317→
   318→  const branchRooms = getRoomsForBranch(branchName)
   319→  let grandTotalAmount = 0
   320→  let grandTotalCars = 0
   321→  let grandTotalNet = 0
   322→
   323→  // Order rooms: regular rooms first, machine last
   324→  const regularRooms = branchRooms.filter(r => r !== 'مكينة الغسيل')
   325→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   326→  const orderedRooms = [...regularRooms]
   327→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   328→
   329→  // Auto-adaptive: calculate GLOBAL size level based on total rooms
   330→  // 0 = normal (<=5 rooms), 1 = compact (never used currently), 2 = ultra-compact (6+)
   331→  const totalRoomCount = orderedRooms.length
   332→  const globalSizeLevel = totalRoomCount <= 5 ? 0 : 2
   333→
   334→  // Build room data with global adaptive sizing
   335→  const buildRoomCells = (sizeLevel: number) => {
   336→    const cells: string[] = []
   337→    orderedRooms.forEach(room => {
   338→      const roomEntries = roomMap[room] || []
   339→      if (roomEntries.length > 0) {
   340→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   341→        const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
   342→        grandTotalAmount += roomTotal
   343→        grandTotalCars += roomCars
   344→        grandTotalNet += getNetAmount(roomTotal, branchName, room)
   345→        cells.push(buildRoomTableHTML(room, roomEntries, branchName, sizeLevel))
   346→      } else {
   347→        cells.push(buildEmptyRoomTableHTML(room, sizeLevel))
   348→      }
   349→    })
   350→    return cells
   351→  }
   352→
   353→  // Build ALL room cells with global sizeLevel
   354→  const roomCells = buildRoomCells(globalSizeLevel)
   355→
   356→  // Helper: build rooms grid HTML from array of room cells (adaptive)
   357→  const buildRoomsGrid = (cells: string[], sizeLevel?: number) => {
   358→    let html = ''
   359→    const sl = sizeLevel || 0
   360→    const gapMap = ['10px', '7px', '5px']
   361→    const mbMap = ['10px', '7px', '5px']
   362→    const gap = gapMap[sl]
   363→    const mb = mbMap[sl]
   364→    for (let i = 0; i < cells.length; i += 2) {
   365→      html += '<div style="display:flex;gap:' + gap + ';margin-bottom:' + mb + ';">' +
   366→        '<div style="flex:1;min-width:0;">' + cells[i] + '</div>' +
   367→        (cells[i + 1] ? '<div style="flex:1;min-width:0;">' + cells[i + 1] + '</div>' : '') +
   368→        '</div>'
   369→    }
   370→    return html
   371→  }
   372→
   373→  // Helper: build page header (adaptive)
   374→  const buildHeader = (sizeLevel?: number) => {
   375→    const sl = sizeLevel || 0
   376→    const fsMap = ['22px', '18px', '15px']
   377→    const dateFsMap = ['14px', '12px', '10px']
   378→    const mbMap = ['14px', '10px', '6px']
   379→    const pbMap = ['10px', '6px', '4px']
   380→    return '<div style="text-align:center;margin-bottom:' + mbMap[sl] + ';border-bottom:2px solid #000;padding-bottom:' + pbMap[sl] + ';">' +
   381→      '<h1 style="font-size:' + fsMap[sl] + ';font-weight:bold;margin:0 0 6px 0;">مغسلة <span style="margin:0 0 0 15px;">' + branchName + '</span></h1>' +
   382→      '<p style="font-size:' + dateFsMap[sl] + ';margin:0;color:#555;">التاريخ: ' + formatDateShort(selectedDate) + '</p>' +
   383→      '</div>'
   384→  }
   385→
   386→  // Adaptive page container padding
   387→  const roomPagePadMap = ['12px 10px', '10px 8px', '8px 6px']
   388→  const treasuryPagePadMap = ['8px 10px', '6px 8px', '4px 6px']
   389→  const sl = globalSizeLevel
   390→
   391→  // Split rooms across pages: max 6 rooms per page (3 rows × 2 cols)
   392→  const MAX_ROOMS_PER_PAGE = 6
   393→  const pages: string[] = []
   394→
   395→  const totalFullPages = Math.floor(roomCells.length / MAX_ROOMS_PER_PAGE)
   396→  const hasRemainder = roomCells.length % MAX_ROOMS_PER_PAGE > 0
   397→
   398→  // Full room pages - ALL use global adaptive sizeLevel
   399→  for (let p = 0; p < totalFullPages; p++) {
   400→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   401→    const isFullPage = pageRooms.length === MAX_ROOMS_PER_PAGE
   402→    // Full pages (e.g. 6 rooms): no min-height/centering to prevent last table cut-off
   403→    // Partial pages: keep min-height for clean layout
   404→    const containerStyle = isFullPage
   405→      ? 'width:780px;background:#fff;color:#000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   406→      : 'width:780px;min-height:1120px;background:#fff;color:#000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   407→    const contentStyle = isFullPage
   408→      ? 'display:flex;flex-direction:column;align-items:center;'
   409→      : 'flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;'
   410→    pages.push(
   411→      '<div style="' + containerStyle + '" dir="rtl">' +
   412→      buildHeader(sl) +
   413→      '<div style="' + contentStyle + '">' +
   414→      buildRoomsGrid(pageRooms, sl) +
   415→      '</div>' +
   416→      '</div>'
   417→    )
   418→  }
   419→
   420→  // Overflow rooms for treasury page
   421→  const overflowRoomCount = hasRemainder ? roomCells.length % MAX_ROOMS_PER_PAGE : 0
   422→  const overflowCells = overflowRoomCount > 0 ? roomCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE) : []
   423→
   424→  // Treasury page - uses same global sizeLevel
   425→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses, sl)
   426→  const overflowHtml = overflowCells.length > 0 ? buildRoomsGrid(overflowCells, sl) : ''
   427→  pages.push(
   428→    '<div style="width:780px;background:#fff;color:#000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   429→    buildHeader(sl) +
   430→    overflowHtml +
   431→    treasuryContent +
   432→    '</div>'
   433→  )
   434→
   435→  return pages
   436→}

   566→    '</div>'
   567→
   568→  return '<div style="' + sepMtMap[sl] + 'border-top:2px solid #000;"><div style="display:flex;' + sepGapMap[sl] + '">' + workerExpensesHtml + treasuryHtml + '</div></div>'

   732→    pages.push('<div style="' + (isLastPage ? pageWithMinHeight : pageStyle) + '">' + currentPageContent + '</div>')
   733→  }
   734→
   735→  return pages
