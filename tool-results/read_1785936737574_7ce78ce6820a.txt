   115→   115→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   116→   116→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   117→   117→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   118→   118→  return Math.floor(totalAmount / 2) - deduction

   311→   311→      const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)

   318→   318→      roomCells.push(buildEmptyRoomTableHTML(room))
   319→   319→    }
   320→   320→  })

   329→   329→      } else {
   330→   330→        cells.push(buildEmptyRoomTableHTML(room))

   379→   379→

   385→   385→
   386→   386→  // Treasury page (always last) - includes overflow rooms if any
   387→   387→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses)

   394→   394→    '</div>'
   395→   395→  )
   396→   396→
