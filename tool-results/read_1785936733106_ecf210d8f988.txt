   113→function getRoomsForBranch(branchName: string) { return BRANCH_ROOMS[branchName] || ROOMS }
   114→function getPricesForRoom(room: string) { return ROOM_PRICES[room] || ALL_PRICES }
   115→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   116→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   117→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   118→  return Math.floor(totalAmount / 2) - deduction
   119→}
   120→function getNetFormulaText(branchName: string, roomName: string) {
   121→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return 'الإجمالي ÷ 2'
   122→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   123→  if (deduction === 0) return 'الإجمالي ÷ 2'
   124→  return `الإجمالي ÷ 2 - ${deduction}`
   125→}
   126→function todayISO() {
   127→  const d = new Date()
   128→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   129→}
   130→function formatDateShort(dateStr: string) {
   131→  if (!dateStr) return ''
   132→  const [y, m, d] = dateStr.split('-')
   133→  return `${d}/${m}/${y}`
   134→}

   137→function buildRoomTableHTML(room: string, roomEntries: CarEntry[], branchName: string, compact?: boolean) {
   138→  const prices = getPricesForRoom(room)
   139→  let roomTotalAmount = 0
   140→  let roomTotalCars = 0
   141→  let roomExtraCars = 0
   142→  let roomExtraAmount = 0
   143→  const mergedCounts: Record<string, number> = {}
   144→  const mergedCustoms: Record<string, { price: number; count: number }> = {}
   145→
   146→  roomEntries.forEach(entry => {
   147→    prices.forEach(price => {
   148→      const count = (entry.priceCounts && entry.priceCounts[price]) || 0
   149→      mergedCounts[price] = (mergedCounts[price] || 0) + count
   150→    })
   151→    if (entry.customPrices) {
   152→      Object.keys(entry.customPrices).forEach(key => {
   153→        const item = entry.customPrices[key]
   154→        if (mergedCustoms[key]) {
   155→          mergedCustoms[key].count += item.count
   156→        } else {
   157→          mergedCustoms[key] = { price: item.price, count: item.count }
   158→        }
   159→      })
   160→    }
   161→    roomTotalAmount += entry.totalAmount
   162→    roomTotalCars += entry.totalCars
   163→    roomExtraCars += (entry.extraCars || 0)
   164→    roomExtraAmount += (entry.extraAmount || 0)
   165→  })
   166→
   167→  const roomNet = getNetAmount(roomTotalAmount, branchName, room)
   168→  const _pad = compact ? '5px 8px' : '10px 10px'
   169→  const cellPad = 'padding:' + _pad + ';vertical-align:middle;'
   170→  const cellFs = compact ? 'font-size:9px;' : 'font-size:10px;'
   171→  const titlePad = compact ? 'padding:6px 8px;' : 'padding:10px 10px;'
   172→  const titleFs = compact ? 'font-size:10px;' : 'font-size:11px;'
   173→
   174→  let rowsHtml = ''
   175→  let rowNum = 0
   176→  prices.forEach((price, idx) => {
   177→    const count = mergedCounts[price] || 0
   178→    rowNum++

   186→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + '">' + rowAmount + ' د.ل</td>' +
   187→    '</tr>'
   188→  })
   189→
   190→  const customKeys = Object.keys(mergedCustoms)
   191→  customKeys.forEach(key => {
   192→    const item = mergedCustoms[key]
   193→    rowNum++
   194→    rowsHtml += '<tr>' +
   195→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + 'color:#7c3aed;">✦</td>' +
   196→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + 'color:#7c3aed;">' + item.price + ' د.ل</td>' +
   197→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;font-size:11px;font-weight:bold;">' + item.count + '</td>' +
   198→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + 'font-weight:bold;">' + (item.price * item.count) + ' د.ل</td>' +
   199→    '</tr>'
   200→  })
   201→
   202→  let extraRowHtml = ''
   203→  if (roomExtraCars > 0) {
   204→    extraRowHtml = '<tr style="background:#fffde7;">' +
   205→      '<td colspan="2" style="' + cellPad + 'border:1px solid #555;font-size:10px;font-weight:bold;text-align:center;color:#e65100;">⭐ إكسترا</td>' +
   206→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;font-size:11px;font-weight:bold;color:#e65100;">' + roomExtraCars + '</td>' +
   207→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;font-size:11px;font-weight:bold;color:#e65100;">' + roomExtraAmount + ' د.ل</td>' +
   208→    '</tr>'
   209→  }
   210→
   211→  return '<table style="width:100%;border-collapse:collapse;font-family:Cairo,sans-serif;table-layout:fixed;border:1px solid #333;">' +
   212→    '<colgroup><col style="width:10%;"/><col style="width:28%;"/><col style="width:24%;"/><col style="width:38%;"/></colgroup>' +
   213→    '<tr><td colspan="4" style="' + titlePad + 'text-align:center;' + titleFs + 'font-weight:bold;background:#bdbdbd;color:#222;">' + room + '</td></tr>' +
   214→    '<tr style="background:#e0e0e0;">' +
   215→    '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + 'font-weight:bold;">م</td>' +

   287→function buildCarReportHTML(selectedDate: string, branchId: string, branchName: string, entries: CarEntry[], savedWorkerExpenses?: Record<string, { cleanliness?: number; treasury?: Record<string, { income: number; expense: number }> }>): string[] {
   288→  // Group entries by room
   289→  const roomMap: Record<string, CarEntry[]> = {}
   290→  entries.forEach(e => {
   291→    if (!roomMap[e.room]) roomMap[e.room] = []
   292→    roomMap[e.room].push(e)
   293→  })
   294→
   295→  const branchRooms = getRoomsForBranch(branchName)
   296→  let grandTotalAmount = 0
   297→  let grandTotalCars = 0
   298→  let grandTotalNet = 0
   299→
   300→  // Order rooms: regular rooms first, machine last
   301→  const regularRooms = branchRooms.filter(r => r !== 'مكينة الغسيل')
   302→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   303→  const orderedRooms = [...regularRooms]
   304→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   305→
   306→  // Build room data (normal - for first page)
   307→  const roomCells: string[] = []
   308→  orderedRooms.forEach(room => {
   309→    const roomEntries = roomMap[room] || []
   310→    if (roomEntries.length > 0) {
   311→      const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   312→      const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
   313→      grandTotalAmount += roomTotal
   314→      grandTotalCars += roomCars
   315→      grandTotalNet += getNetAmount(roomTotal, branchName, room)
   316→      roomCells.push(buildRoomTableHTML(room, roomEntries, branchName))
   317→    } else {
   318→      roomCells.push(buildEmptyRoomTableHTML(room))
   319→    }
   320→  })
   321→
   322→  // Build compact room cells (for overflow/second page)
   323→  const buildCompactRoomCells = () => {
   324→    const cells: string[] = []
   325→    orderedRooms.forEach(room => {
   326→      const roomEntries = roomMap[room] || []
   327→      if (roomEntries.length > 0) {
   328→        cells.push(buildRoomTableHTML(room, roomEntries, branchName, true))
   329→      } else {
   330→        cells.push(buildEmptyRoomTableHTML(room))
   331→      }
   332→    })
   333→    return cells
   334→  }
   335→
   336→  // Helper: build rooms grid HTML from array of room cells
   337→  const buildRoomsGrid = (cells: string[], compact?: boolean) => {
   338→    let html = ''
   339→    const gap = compact ? '5px' : '8px'
   340→    const mb = compact ? '4px' : '6px'
   341→    for (let i = 0; i < cells.length; i += 2) {
   342→      html += '<div style="display:flex;gap:' + gap + ';margin-bottom:' + mb + ';">' +
   343→        '<div style="flex:1;min-width:0;">' + cells[i] + '</div>' +
   344→        (cells[i + 1] ? '<div style="flex:1;min-width:0;">' + cells[i + 1] + '</div>' : '') +
   345→        '</div>'
   346→    }
   347→    return html
   348→  }
   349→
   350→  // Helper: build page header
   351→  const buildHeader = (title: string) => {
   352→    return '<div style="text-align:center;margin-bottom:14px;border-bottom:2px solid #000;padding-bottom:10px;">' +
   353→      '<h1 style="font-size:22px;font-weight:bold;margin:0 0 10px 0;">مغسلة <span style="margin:0 0 0 15px;">' + branchName + '</span></h1>' +
   354→      '<p style="font-size:14px;margin:0;color:#555;">التاريخ: ' + formatDateShort(selectedDate) + '</p>' +
   355→      '</div>'
   356→  }
   357→
   358→  // Split rooms across pages: max 4 rooms per page (2 rows × 2 cols)
   359→  const MAX_ROOMS_PER_PAGE = 4
   360→  const pages: string[] = []
   361→
   362→  // Room pages - if last room page is not full, merge it with treasury page
   363→  let overflowRooms: string[] = []
   364→  const totalFullPages = Math.floor(roomCells.length / MAX_ROOMS_PER_PAGE)
   365→  const hasRemainder = roomCells.length % MAX_ROOMS_PER_PAGE > 0
   366→
   367→  // Full room pages
   368→  for (let p = 0; p < totalFullPages; p++) {
   369→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   370→    pages.push(
   371→      '<div style="width:780px;min-height:1120px;background:#fff;color:#000;padding:12px 10px;font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;" dir="rtl">' +
   372→      buildHeader('تقرير تسجيل السيارات') +
   373→      '<div style="flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;">' +
   374→      buildRoomsGrid(pageRooms) +
   375→      '</div>' +
   376→      '</div>'
   377→    )
   378→  }
   379→
   380→  // Remaining rooms go to treasury page (compact)
   381→  if (hasRemainder) {
   382→    const compactCells = buildCompactRoomCells()
   383→    overflowRooms = compactCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE)
   384→  }
   385→
   386→  // Treasury page (always last) - includes overflow rooms if any
   387→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses)
   388→  const overflowHtml = overflowRooms.length > 0 ? buildRoomsGrid(overflowRooms, true) : ''
   389→  pages.push(
   390→    '<div style="width:780px;background:#fff;color:#000;padding:8px 10px;font-family:Cairo,sans-serif;" dir="rtl">' +
   391→    buildHeader('مصاريف العمال والخزينة') +
   392→    overflowHtml +
   393→    treasuryContent +
   394→    '</div>'
   395→  )
   396→
   397→  return pages
   398→}

   400→// Helper: build worker expenses + treasury section (used on last page of car report)
   401→function buildWorkerExpensesAndTreasury(
   402→  branchName: string, selectedDate: string, orderedRooms: string[], entries: CarEntry[],
   403→  grandTotalNet: number, savedWorkerExpenses?: Record<string, { cleanliness?: number; treasury?: Record<string, { income: number; expense: number }> }>
   404→): string {
   405→  const cleannessConfig = BRANCH_CLEANLINESS[branchName]
   406→  const wKey = branchName + '_' + selectedDate
   407→  const savedWE = savedWorkerExpenses?.[wKey] || {}
   408→  let savedCleanliness = 0
   409→  if (cleannessConfig) {
