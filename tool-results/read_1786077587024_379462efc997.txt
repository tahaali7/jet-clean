   313→function buildCarReportHTML(selectedDate: string, branchId: string, branchName: string, entries: CarEntry[], savedWorkerExpenses?: Record<string, { cleanliness?: number; treasury?: Record<string, { income: number; expense: number }> }>): string[] {
   314→  // Group entries by room
   315→  const roomMap: Record<string, CarEntry[]> = {}
   316→  entries.forEach(e => {
   317→    if (!roomMap[e.room]) roomMap[e.room] = []
   318→    roomMap[e.room].push(e)
   319→  })
   320→
   321→  const branchRooms = getRoomsForBranch(branchName)
   322→  let grandTotalAmount = 0
   323→  let grandTotalCars = 0
   324→  let grandTotalNet = 0
   325→
   326→  // Order rooms: regular rooms first, machine last
   327→  const regularRooms = branchRooms.filter(r => r !== 'مكينة الغسيل')
   328→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   329→  const orderedRooms = [...regularRooms]
   330→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   331→
   332→  // Auto-adaptive: calculate GLOBAL size level based on total rooms
   333→  // 0 = normal (<=5 rooms), 1 = compact (never used currently), 2 = ultra-compact (6+)
   334→  const totalRoomCount = orderedRooms.length
   335→  const globalSizeLevel = totalRoomCount <= 5 ? 0 : 2
   336→
   337→  // Build room data with global adaptive sizing
   338→  const buildRoomCells = (sizeLevel: number) => {
   339→    const cells: string[] = []
   340→    orderedRooms.forEach(room => {
   341→      const roomEntries = roomMap[room] || []
   342→      if (roomEntries.length > 0) {
   343→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   344→        const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
   345→        grandTotalAmount += roomTotal
   346→        grandTotalCars += roomCars
   347→        grandTotalNet += getNetAmount(roomTotal, branchName, room)
   348→        cells.push(buildRoomTableHTML(room, roomEntries, branchName, sizeLevel))
   349→      } else {
   350→        cells.push(buildEmptyRoomTableHTML(room, sizeLevel))
   351→      }
   352→    })
   353→    return cells
   354→  }
   355→
   356→  // Build ALL room cells with global sizeLevel
   357→  const roomCells = buildRoomCells(globalSizeLevel)
   358→
   359→  // Helper: build rooms grid HTML from array of room cells (adaptive)
   360→  const buildRoomsGrid = (cells: string[], sizeLevel?: number) => {
   361→    let html = ''
   362→    const sl = sizeLevel || 0
   363→    const gapMap = ['10px', '7px', '5px']
   364→    const mbMap = ['10px', '7px', '5px']
   365→    const gap = gapMap[sl]
   366→    const mb = mbMap[sl]
   367→    for (let i = 0; i < cells.length; i += 2) {
   368→      html += '<div style="display:flex;gap:' + gap + ';margin-bottom:' + mb + ';">' +
   369→        '<div style="flex:1;min-width:0;">' + cells[i] + '</div>' +
   370→        (cells[i + 1] ? '<div style="flex:1;min-width:0;">' + cells[i + 1] + '</div>' : '') +
   371→        '</div>'
   372→    }
   373→    return html
   374→  }
   375→
   376→  // Helper: build page header (adaptive)
   377→  const buildHeader = (sizeLevel?: number) => {
   378→    const sl = sizeLevel || 0
   379→    const fsMap = ['22px', '18px', '15px']
   380→    const dateFsMap = ['14px', '12px', '10px']
   381→    const mbMap = ['14px', '10px', '6px']
   382→    const pbMap = ['10px', '6px', '4px']
   383→    return '<div style="text-align:center;margin-bottom:' + mbMap[sl] + ';border-bottom:2px solid #000;padding-bottom:' + pbMap[sl] + ';">' +
   384→      '<h1 style="font-size:' + fsMap[sl] + ';font-weight:bold;margin:0 0 6px 0;">مغسلة <span style="margin:0 0 0 15px;">' + branchName + '</span></h1>' +
   385→      '<p style="font-size:' + dateFsMap[sl] + ';margin:0;color:#000000;">التاريخ: ' + formatDateShort(selectedDate) + '</p>' +
   386→      '</div>'
   387→  }
   388→
   389→  // Adaptive page container padding
   390→  const roomPagePadMap = ['12px 10px', '10px 8px', '8px 6px']
   391→  const treasuryPagePadMap = ['8px 10px', '6px 8px', '4px 6px']
   392→  const sl = globalSizeLevel
   393→
   394→  // Split rooms across pages: max 6 rooms per page (3 rows × 2 cols)
   395→  const MAX_ROOMS_PER_PAGE = 6
   396→  const pages: string[] = []
   397→
   398→  const totalFullPages = Math.floor(roomCells.length / MAX_ROOMS_PER_PAGE)
   399→  const hasRemainder = roomCells.length % MAX_ROOMS_PER_PAGE > 0
   400→
   401→  // Full room pages - ALL use global adaptive sizeLevel
   402→  for (let p = 0; p < totalFullPages; p++) {
   403→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   404→    const isFullPage = pageRooms.length === MAX_ROOMS_PER_PAGE
   405→    // Full pages (e.g. 6 rooms): no min-height/centering to prevent last table cut-off
   406→    // Partial pages: keep min-height for clean layout
   407→    const containerStyle = isFullPage
   408→      ? 'width:780px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   409→      : 'width:780px;min-height:1120px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   410→    const contentStyle = isFullPage
   411→      ? 'display:flex;flex-direction:column;align-items:center;'
   412→      : 'flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;'
   413→    pages.push(
   414→      '<div style="' + containerStyle + '" dir="rtl">' +
   415→      buildHeader(sl) +
   416→      '<div style="' + contentStyle + '">' +
   417→      buildRoomsGrid(pageRooms, sl) +
   418→      '</div>' +
   419→      '</div>'
   420→    )
   421→  }
   422→
   423→  // Overflow rooms for treasury page
   424→  const overflowRoomCount = hasRemainder ? roomCells.length % MAX_ROOMS_PER_PAGE : 0
   425→  const overflowCells = overflowRoomCount > 0 ? roomCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE) : []
   426→
   427→  // Treasury page - uses same global sizeLevel
   428→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses, sl)
   429→  const overflowHtml = overflowCells.length > 0 ? buildRoomsGrid(overflowCells, sl) : ''
   430→  pages.push(
   431→    '<div style="width:780px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   432→    buildHeader(sl) +
   433→    overflowHtml +
   434→    treasuryContent +
   435→    '</div>'
   436→  )
   437→
   438→  return pages
   439→}

   808→}
