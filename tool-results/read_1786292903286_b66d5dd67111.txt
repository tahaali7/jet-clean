   121→function getRoomsForBranch(branchName: string) { return BRANCH_ROOMS[branchName] || ROOMS }
   122→function getPricesForRoom(room: string) { return ROOM_PRICES[room] || ALL_PRICES }
   123→function isExtraEnabledForBranch(branchName: string) { return !BRANCH_EXTRA_DISABLED.includes(branchName) }
   124→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   125→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   126→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   127→  return Math.floor(totalAmount / 2) - deduction
   128→}
   129→function getNetFormulaText(branchName: string, roomName: string) {
   130→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return 'الإجمالي ÷ 2'
   131→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   132→  if (deduction === 0) return 'الإجمالي ÷ 2'
   133→  return `الإجمالي ÷ 2 - ${deduction}`
   134→}
   135→function todayISO() {
   136→  const d = new Date()
   137→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   138→}
   139→function formatDateShort(dateStr: string) {
   140→  if (!dateStr) return ''
   141→  const [y, m, d] = dateStr.split('-')
   142→  return `${d}/${m}/${y}`
   143→}

   153→  const mergedCounts: Record<string, number> = {}
   154→  const mergedCustoms: Record<string, { price: number; count: number }> = {}
   155→
   156→  roomEntries.forEach(entry => {
   157→    prices.forEach(price => {
   158→      const count = (entry.priceCounts && entry.priceCounts[price]) || 0
   159→      mergedCounts[price] = (mergedCounts[price] || 0) + count
   160→    })
   161→    if (entry.customPrices) {
   162→      Object.keys(entry.customPrices).forEach(key => {
   163→        const item = entry.customPrices[key]
   164→        if (mergedCustoms[key]) {
   165→          mergedCustoms[key].count += item.count
   166→        } else {
   167→          mergedCustoms[key] = { price: item.price, count: item.count }
   168→        }
   169→      })
   170→    }
   171→    roomTotalAmount += entry.totalAmount
   172→    roomTotalCars += entry.totalCars
   173→    roomExtraCars += (entry.extraCars || 0)
   174→    roomExtraAmount += (entry.extraAmount || 0)

   177→  const roomNet = getNetAmount(roomTotalAmount, branchName, room)
   178→  // Auto-adaptive: level 0=normal, 1=compact, 2=ultra-compact
   179→  const sl = sizeLevel || 0
   180→  const padMap = ['10px 10px', '5px 7px', '3px 5px']

   205→    '</tr>'

   315→function buildCarReportHTML(selectedDate: string, branchId: string, branchName: string, entries: CarEntry[], savedWorkerExpenses?: Record<string, { cleanliness?: number; treasury?: Record<string, { income: number; expense: number }> }>): string[] {
   316→  // Group entries by room
   317→  const roomMap: Record<string, CarEntry[]> = {}
   318→  entries.forEach(e => {
   319→    if (!roomMap[e.room]) roomMap[e.room] = []
   320→    roomMap[e.room].push(e)
   321→  })
   322→
   323→  const branchRooms = getRoomsForBranch(branchName)
   324→  let grandTotalAmount = 0
   325→  let grandTotalCars = 0
   326→  let grandTotalNet = 0
   327→
   328→  // Order rooms: regular rooms first, machine last
   329→  const regularRooms = branchRooms.filter(r => r !== 'مكينة الغسيل')
   330→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   331→  const orderedRooms = [...regularRooms]
   332→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   333→
   334→  // Auto-adaptive: calculate GLOBAL size level based on total rooms
   335→  // 0 = normal (<=5 rooms), 1 = compact (never used currently), 2 = ultra-compact (6+)
   336→  const totalRoomCount = orderedRooms.length
   337→  const globalSizeLevel = totalRoomCount <= 5 ? 0 : 2
   338→
   339→  // Build room data with global adaptive sizing
   340→  const buildRoomCells = (sizeLevel: number) => {
   341→    const cells: string[] = []
   342→    orderedRooms.forEach(room => {
   343→      const roomEntries = roomMap[room] || []
   344→      if (roomEntries.length > 0) {
   345→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   346→        const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
   347→        grandTotalAmount += roomTotal
   348→        grandTotalCars += roomCars
   349→        grandTotalNet += getNetAmount(roomTotal, branchName, room)
   350→        cells.push(buildRoomTableHTML(room, roomEntries, branchName, sizeLevel))
   351→      } else {
   352→        cells.push(buildEmptyRoomTableHTML(room, sizeLevel))
   353→      }
   354→    })
   355→    return cells
   356→  }
   357→
   358→  // Build ALL room cells with global sizeLevel
   359→  const roomCells = buildRoomCells(globalSizeLevel)
   360→
   361→  // Helper: build rooms grid HTML from array of room cells (adaptive)
   362→  const buildRoomsGrid = (cells: string[], sizeLevel?: number) => {
   363→    let html = ''
   364→    const sl = sizeLevel || 0
   365→    const gapMap = ['10px', '7px', '5px']
   366→    const mbMap = ['10px', '7px', '5px']

   368→    const mb = mbMap[sl]

   371→        '<div style="flex:1;min-width:0;">' + cells[i] + '</div>' +
   372→        (cells[i + 1] ? '<div style="flex:1;min-width:0;">' + cells[i + 1] + '</div>' : '') +
   373→        '</div>'
   374→    }
   375→    return html
   376→  }
   377→
   378→  // Helper: build page header (adaptive)
   379→  const buildHeader = (sizeLevel?: number) => {
   380→    const sl = sizeLevel || 0
   381→    const fsMap = ['22px', '18px', '15px']
   382→    const dateFsMap = ['14px', '12px', '10px']
   383→    const mbMap = ['14px', '10px', '6px']
   384→    const pbMap = ['10px', '6px', '4px']
   385→    return '<div style="text-align:center;margin-bottom:' + mbMap[sl] + ';border-bottom:2px solid #000;padding-bottom:' + pbMap[sl] + ';">' +
   386→      '<h1 style="font-size:' + fsMap[sl] + ';font-weight:bold;margin:0 0 6px 0;">مغسلة <span style="margin:0 0 0 15px;">' + branchName + '</span></h1>' +
   387→      '<p style="font-size:' + dateFsMap[sl] + ';margin:0;color:#000000;">التاريخ: ' + formatDateShort(selectedDate) + '</p>' +
   388→      '</div>'
   389→  }
   390→
   391→  // Adaptive page container padding
   392→  const roomPagePadMap = ['12px 10px', '10px 8px', '8px 6px']
   393→  const treasuryPagePadMap = ['8px 10px', '6px 8px', '4px 6px']
   394→  const sl = globalSizeLevel
   395→
   396→  // Split rooms across pages: max 6 rooms per page (3 rows × 2 cols)
   397→  const MAX_ROOMS_PER_PAGE = 6
   398→  const pages: string[] = []
   399→
   400→  const totalFullPages = Math.floor(roomCells.length / MAX_ROOMS_PER_PAGE)
   401→  const hasRemainder = roomCells.length % MAX_ROOMS_PER_PAGE > 0
   402→
   403→  // Full room pages - ALL use global adaptive sizeLevel
   404→  for (let p = 0; p < totalFullPages; p++) {
   405→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   406→    const isFullPage = pageRooms.length === MAX_ROOMS_PER_PAGE
   407→    // Full pages (e.g. 6 rooms): no min-height/centering to prevent last table cut-off
   408→    // Partial pages: keep min-height for clean layout
   409→    const containerStyle = isFullPage
   410→      ? 'width:780px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   411→      : 'width:780px;min-height:1120px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   412→    const contentStyle = isFullPage
   413→      ? 'display:flex;flex-direction:column;align-items:center;'
   414→      : 'flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;'
   415→    pages.push(
   416→      '<div style="' + containerStyle + '" dir="rtl">' +
   417→      buildHeader(sl) +
   418→      '<div style="' + contentStyle + '">' +
   419→      buildRoomsGrid(pageRooms, sl) +
   420→      '</div>' +
   421→      '</div>'
   422→    )
   423→  }
   424→
   425→  // Overflow rooms for treasury page
   426→  const overflowRoomCount = hasRemainder ? roomCells.length % MAX_ROOMS_PER_PAGE : 0
   427→  const overflowCells = overflowRoomCount > 0 ? roomCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE) : []
   428→
   429→  // Treasury page - uses same global sizeLevel
   430→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses, sl)
   431→  const overflowHtml = overflowCells.length > 0 ? buildRoomsGrid(overflowCells, sl) : ''
   432→  pages.push(
   433→    '<div style="width:780px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   434→    buildHeader(sl) +
   435→    overflowHtml +
   436→    treasuryContent +
   437→    '</div>'
   438→  )
   439→
   440→  return pages
   441→}

   754→  const footerHtml = '<div style="text-align:center;margin-top:12px;padding-top:8px;border-top:1px solid #e2e8f0;font-size:9px;color:#000000;">صفحة __PAGE__</div>'
   755→  const pageStyle = 'width:800px;color:#000000;padding:24px;font-family:Cairo,sans-serif;'
   756→  const pageWithMinHeight = pageStyle + 'min-height:1120px;'
   757→
   758→  // Render each branch as a separate block, then pack into pages
   759→  const branchBlocks: { html: string; estimatedHeight: number }[] = []
   760→  branchDatas.forEach(bd => {
   761→    const blockHtml = '<div style="margin-bottom:16px;">' +
   762→      '<h3 style="background:#0e7490;color:#fff;padding:6px 10px;border-radius:6px;font-size:13px;margin-bottom:6px;">' +
   763→      'فرع ' + bd.name + ' — سحوبات: ' + bd.withdrawals + ' د.ل | عجوزات: ' + bd.shortages + ' د.ل | الإجمالي: ' + (bd.withdrawals + bd.shortages) + ' د.ل' +

   771→      '</tr></thead>' +
   772→      '<tbody>' + bd.empsHtml + '</tbody>' +
   773→      '</table></div>'
   774→    // Estimate: title ~30px + header row ~30px + each emp row ~35px + detail rows ~25px each
   775→    const rowCount = bd.empsHtml.split('<tr').length
   776→    branchBlocks.push({ html: blockHtml, estimatedHeight: 30 + 30 + rowCount * 35 })
   777→  })
   778→
   779→  // Header + summary estimated ~200px, footer ~30px, page usable ~880px (1120 - 200 - 30)
   780→  const headerHeight = 200
   781→  const footerHeight = 30
   782→  const usableHeight = 890
   783→
   784→  let pageNum = 1
   785→  let currentPageContent = headerHtml + summaryHtml
   786→  let currentPageUsed = headerHeight
   787→
   788→  branchBlocks.forEach((block, idx) => {
   789→    if (currentPageUsed + block.estimatedHeight + footerHeight > 1120 && idx > 0) {
   790→      // Current page is full, start a new one
   791→      currentPageContent += footerHtml.replace('__PAGE__', String(pageNum))
   792→      pages.push('<div style="' + pageWithMinHeight + '">' + currentPageContent + '</div>')
   793→      pageNum++
   794→      currentPageContent = headerHtml + block.html
   795→      currentPageUsed = headerHeight + block.estimatedHeight
   796→    } else {
   797→      currentPageContent += block.html
   798→      currentPageUsed += block.estimatedHeight
   799→    }
   800→  })
   801→
   802→  // Last page
   803→  if (currentPageContent) {
   804→    currentPageContent += footerHtml.replace('__PAGE__', String(pageNum))
   805→    const isLastPage = branchBlocks.length <= 2
   806→    pages.push('<div style="' + (isLastPage ? pageWithMinHeight : pageStyle) + '">' + currentPageContent + '</div>')
   807→  }
   808→
   809→  return pages
   810→}
