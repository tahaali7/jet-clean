   113→function getRoomsForBranch(branchName: string) { return BRANCH_ROOMS[branchName] || ROOMS }
   114→function getPricesForRoom(room: string) { return ROOM_PRICES[room] || ALL_PRICES }
   115→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   116→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   117→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   118→  return Math.floor(totalAmount / 2) - deduction
   119→}
   120→function getNetFormulaText(branchName: string, roomName: string) {
   121→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return 'الإجمالي ÷ 2'
   122→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   123→  if (deduction === 0) return 'الإجمالي ÷ 2'
   124→  return `الإجمالي ÷ 2 - ${deduction}`
   125→}
   126→function todayISO() {
   127→  const d = new Date()
   128→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   129→}
   130→function formatDateShort(dateStr: string) {
   131→  if (!dateStr) return ''
   132→  const [y, m, d] = dateStr.split('-')
   133→  return `${d}/${m}/${y}`
   134→}

   137→function buildRoomTableHTML(room: string, roomEntries: CarEntry[], branchName: string) {
   138→  const prices = getPricesForRoom(room)
   139→  let roomTotalAmount = 0
   140→  let roomTotalCars = 0
   141→  const mergedCounts: Record<string, number> = {}
   142→  const mergedCustoms: Record<string, { price: number; count: number }> = {}
   143→
   144→  roomEntries.forEach(entry => {
   145→    prices.forEach(price => {
   146→      const count = (entry.priceCounts && entry.priceCounts[price]) || 0
   147→      mergedCounts[price] = (mergedCounts[price] || 0) + count
   148→    })
   149→    if (entry.customPrices) {
   150→      Object.keys(entry.customPrices).forEach(key => {
   151→        const item = entry.customPrices[key]
   152→        if (mergedCustoms[key]) {
   153→          mergedCustoms[key].count += item.count
   154→        } else {
   155→          mergedCustoms[key] = { price: item.price, count: item.count }
   156→        }
   157→      })
   158→    }
   159→    roomTotalAmount += entry.totalAmount
   160→    roomTotalCars += entry.totalCars
   161→  })
   162→
   163→  const roomNet = getNetAmount(roomTotalAmount, branchName, room)
   164→  const cellPad = 'padding:5px 4px;vertical-align:middle;'
   165→  const cellFs = 'font-size:10px;'
   166→
   167→  // Build rows - only non-empty
   168→  let rowsHtml = ''
   169→  let rowNum = 0
   170→  prices.forEach(price => {
   171→    const count = mergedCounts[price] || 0
   172→    if (count === 0) return
   173→    rowNum++
   174→    const isExtra = EXTRA_PRICES.includes(price)
   175→    const displayPrice = isExtra ? (price - 5) : price
   176→    const rowAmount = displayPrice * count
   177→    rowsHtml += '<tr>' +
   178→      '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + '">' + rowNum + '</td>' +
   179→      '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + '">' + displayPrice + ' د.ل</td>' +
   180→      '<td style="' + cellPad + 'border:1px solid #333;text-align:center;font-size:11px;font-weight:bold;">' + count + '</td>' +
   181→      '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + '">' + rowAmount + ' د.ل</td>' +
   182→      '</tr>'
   183→  })
   184→
   185→  // Custom price rows
   186→  const customKeys = Object.keys(mergedCustoms)
   187→  customKeys.forEach(key => {
   188→    const item = mergedCustoms[key]
   189→    rowNum++
   190→    rowsHtml += '<tr>' +
   191→      '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + 'color:#7c3aed;">✦</td>' +
   192→      '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + 'color:#7c3aed;">' + item.price + ' د.ل</td>' +
   193→      '<td style="' + cellPad + 'border:1px solid #333;text-align:center;font-size:11px;font-weight:bold;">' + item.count + '</td>' +
   194→      '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + 'font-weight:bold;">' + (item.price * item.count) + ' د.ل</td>' +
   195→      '</tr>'
   196→  })
   197→
   198→  return '<table style="width:100%;border-collapse:collapse;font-family:Cairo,sans-serif;">' +
   199→    '<tr><td colspan="4" style="border:1px solid #333;padding:5px 4px;text-align:center;font-size:12px;font-weight:bold;background:#e8e8e8;vertical-align:middle;">' + room + '</td></tr>' +
   200→    '<tr style="background:#f0f0f0;">' +
   201→    '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + 'font-weight:bold;">م</td>' +
   202→    '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + 'font-weight:bold;">السعر</td>' +
   203→    '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + 'font-weight:bold;">العدد</td>' +
   204→    '<td style="' + cellPad + 'border:1px solid #333;text-align:center;' + cellFs + 'font-weight:bold;">الإجمالي</td>' +
   205→    '</tr>' +
   206→    rowsHtml +
   207→    '<tr style="background:#f0f0f0;">' +
   208→    '<td colspan="2" style="' + cellPad + 'border:1px solid #333;font-size:10px;font-weight:bold;text-align:center;">إجمالي الغرفة</td>' +
   209→    '<td colspan="2" style="' + cellPad + 'border:1px solid #333;text-align:center;font-size:11px;font-weight:bold;">' + roomTotalCars + ' سيارة = ' + roomTotalAmount + ' د.ل</td>' +
   210→    '</tr>' +
   211→    '<tr>' +

   214→    '</tr>' +
   215→    '</table>'

   322→// ==================== EMPLOYEE REPORT (WITHDRAWALS/SHORTAGES) ====================
   323→function buildEmployeeReportHTML(
   324→  periodLabel: string,
   325→  allEmployees: Employee[],
   326→  allRecords: FinancialRecord[],
   327→  allBranches: Branch[],
   328→  matchRecord: (r: FinancialRecord) => boolean
   329→): string {
   330→  const now = new Date()
   331→  const generatedOn = now.toLocaleDateString('ar-LY', { year: 'numeric', month: 'long', day: 'numeric' })
   332→
   333→  let grandWithdrawals = 0
   334→  let grandShortages = 0
   335→  let branchesHtml = ''
   336→
   337→  allBranches.forEach(branch => {
   338→    const branchEmps = allEmployees.filter(e => e.branchId === branch.id)
   339→    if (branchEmps.length === 0) return
   340→
   341→    let branchWithdrawals = 0
   342→    let branchShortages = 0
   343→    let rowsHtml = ''
   344→    let branchHasRecords = false
   345→
   346→    branchEmps.forEach(emp => {
   347→      const empRecords = allRecords
   348→        .filter(r => r.empId === emp.id && matchRecord(r))
   349→        .sort((a, b) => (a.date || '').localeCompare(b.date || ''))
   350→      const withdrawals = empRecords.filter(r => r.type === 'withdrawal').reduce((sum, r) => sum + r.amount, 0)

   352→      branchWithdrawals += withdrawals
   353→      branchShortages += shortages
   354→      grandWithdrawals += withdrawals
   355→      grandShortages += shortages
   356→      if (empRecords.length > 0) branchHasRecords = true
   357→
   358→      const notes = empRecords.map(r =>
   359→        formatDateShort(r.date) + ' — ' + (r.type === 'withdrawal' ? 'سحب' : 'عجز') + ': ' + r.amount + ' د.ل' + (r.note ? ' (' + r.note + ')' : '')

   367→        '<td style="padding:8px;border:1px solid #ddd;font-size:11px;color:#555;">' + notes + '</td>' +
   368→        '</tr>'
   369→    })
   370→
   371→    if (!branchHasRecords) return
   372→
   373→    branchesHtml += '<div style="margin-bottom:24px;">' +
   374→      '<h3 style="background:#0e7490;color:#fff;padding:8px 12px;border-radius:6px;font-size:15px;margin-bottom:8px;">' +
   375→      '📍 فرع: ' + branch.name + ' — إجمالي السحبيات: ' + branchWithdrawals + ' د.ل | إجمالي العجوزات: ' + branchShortages + ' د.ل' +
   376→      '</h3>' +
   377→      '<table style="width:100%;border-collapse:collapse;font-size:12px;">' +
   378→      '<thead><tr style="background:#f1f5f9;">' +
   379→      '<th style="padding:8px;border:1px solid #ddd;">الموظف</th>' +
   380→      '<th style="padding:8px;border:1px solid #ddd;">الفترة</th>' +
   381→      '<th style="padding:8px;border:1px solid #ddd;">السحبيات</th>' +
   382→      '<th style="padding:8px;border:1px solid #ddd;">العجوزات</th>' +
   383→      '<th style="padding:8px;border:1px solid #ddd;">تفاصيل الحركات</th>' +
   384→      '</tr></thead>' +
   385→      '<tbody>' + rowsHtml + '</tbody>' +
   386→      '</table></div>'
   387→  })
