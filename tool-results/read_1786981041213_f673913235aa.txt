   127→function getRoomsForBranch(branchName: string) { return BRANCH_ROOMS[branchName] || ROOMS }
   128→function getPricesForRoom(room: string) { return ROOM_PRICES[room] || ALL_PRICES }
   129→function isExtraEnabledForBranch(branchName: string) { return !BRANCH_EXTRA_DISABLED.includes(branchName) }
   130→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   131→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   132→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   133→  return Math.floor(totalAmount / 2) - deduction
   134→}
   135→function getNetFormulaText(branchName: string, roomName: string) {
   136→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return 'الإجمالي ÷ 2'
   137→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   138→  if (deduction === 0) return 'الإجمالي ÷ 2'
   139→  return `الإجمالي ÷ 2 - ${deduction}`
   140→}
   141→function todayISO() {
   142→  const d = new Date()
   143→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   144→}
   145→function formatDateShort(dateStr: string) {
   146→  if (!dateStr) return ''
   147→  const [y, m, d] = dateStr.split('-')
   148→  return `${d}/${m}/${y}`
   149→}

   156→  let roomTotalCars = 0

   159→  const mergedCounts: Record<string, number> = {}
   160→  const mergedCustoms: Record<string, { price: number; count: number }> = {}
   161→
   162→  roomEntries.forEach(entry => {
   163→    prices.forEach(price => {
   164→      const count = (entry.priceCounts && entry.priceCounts[price]) || 0
   165→      mergedCounts[price] = (mergedCounts[price] || 0) + count
   166→    })
   167→    if (entry.customPrices) {
   168→      Object.keys(entry.customPrices).forEach(key => {
   169→        const item = entry.customPrices[key]
   170→        if (mergedCustoms[key]) {
   171→          mergedCustoms[key].count += item.count
   172→        } else {
   173→          mergedCustoms[key] = { price: item.price, count: item.count }
   174→        }
   175→      })
   176→    }
   177→    roomTotalAmount += entry.totalAmount
   178→    roomTotalCars += entry.totalCars
   179→    roomExtraCars += (entry.extraCars || 0)
   180→    roomExtraAmount += (entry.extraAmount || 0)

   183→  const roomNet = getNetAmount(roomTotalAmount, branchName, room)
   184→  // Auto-adaptive: level 0=normal, 1=compact, 2=ultra-compact

   320→function buildCarReportHTML(selectedDate: string, branchId: string, branchName: string, entries: CarEntry[], savedWorkerExpenses?: Record<string, { cleanliness?: number; treasury?: Record<string, { income: number; expense: number }> }>): string[] {
   321→  // Group entries by room
   322→  const roomMap: Record<string, CarEntry[]> = {}
   323→  entries.forEach(e => {
   324→    if (!roomMap[e.room]) roomMap[e.room] = []
   325→    roomMap[e.room].push(e)
   326→  })
   327→
   328→  const branchRooms = getRoomsForBranch(branchName)
   329→  let grandTotalAmount = 0
   330→  let grandTotalCars = 0
   331→  let grandTotalNet = 0
   332→
   333→  // Order rooms: regular rooms first, machine last
   334→  const regularRooms = branchRooms.filter(r => r !== 'مكينة الغسيل')
   335→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   336→  const orderedRooms = [...regularRooms]
   337→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   338→
   339→  // Auto-adaptive: calculate GLOBAL size level based on total rooms
   340→  // 0 = normal (<=5 rooms), 1 = compact (never used currently), 2 = ultra-compact (6+)
   341→  const totalRoomCount = orderedRooms.length
   342→  const globalSizeLevel = totalRoomCount <= 5 ? 0 : 2
   343→
   344→  // Build room data with global adaptive sizing
   345→  const buildRoomCells = (sizeLevel: number) => {
   346→    const cells: string[] = []
   347→    orderedRooms.forEach(room => {
   348→      const roomEntries = roomMap[room] || []
   349→      if (roomEntries.length > 0) {
   350→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   351→        const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
   352→        grandTotalAmount += roomTotal
   353→        grandTotalCars += roomCars
   354→        grandTotalNet += getNetAmount(roomTotal, branchName, room)
   355→        cells.push(buildRoomTableHTML(room, roomEntries, branchName, sizeLevel))
   356→      } else {
   357→        cells.push(buildEmptyRoomTableHTML(room, sizeLevel))
   358→      }
   359→    })
   360→    return cells
   361→  }
   362→
   363→  // Build ALL room cells with global sizeLevel
   364→  const roomCells = buildRoomCells(globalSizeLevel)
   365→
   366→  // Helper: build rooms grid HTML from array of room cells (adaptive)
   367→  const buildRoomsGrid = (cells: string[], sizeLevel?: number) => {
   368→    let html = ''
   369→    const sl = sizeLevel || 0
   370→    const gapMap = ['10px', '7px', '5px']
   371→    const mbMap = ['10px', '7px', '5px']
   372→    const gap = gapMap[sl]
   373→    const mb = mbMap[sl]
   374→    for (let i = 0; i < cells.length; i += 2) {
   375→      html += '<div style="display:flex;gap:' + gap + ';margin-bottom:' + mb + ';">' +
   376→        '<div style="flex:1;min-width:0;">' + cells[i] + '</div>' +
   377→        (cells[i + 1] ? '<div style="flex:1;min-width:0;">' + cells[i + 1] + '</div>' : '') +
   378→        '</div>'
   379→    }
   380→    return html
   381→  }
   382→
   383→  // Helper: build page header (adaptive)
   384→  const buildHeader = (sizeLevel?: number) => {
   385→    const sl = sizeLevel || 0
   386→    const fsMap = ['22px', '18px', '15px']
   387→    const dateFsMap = ['14px', '12px', '10px']
   388→    const mbMap = ['14px', '10px', '6px']
   389→    const pbMap = ['10px', '6px', '4px']
   390→    return '<div style="text-align:center;margin-bottom:' + mbMap[sl] + ';border-bottom:2px solid #000;padding-bottom:' + pbMap[sl] + ';">' +
   391→      '<h1 style="font-size:' + fsMap[sl] + ';font-weight:bold;margin:0 0 6px 0;">مغسلة <span style="margin:0 0 0 15px;">' + branchName + '</span></h1>' +
   392→      '<p style="font-size:' + dateFsMap[sl] + ';margin:0;color:#000000;">التاريخ: ' + formatDateShort(selectedDate) + '</p>' +
   393→      '</div>'
   394→  }
   395→
   396→  // Adaptive page container padding
   397→  const roomPagePadMap = ['12px 10px', '10px 8px', '8px 6px']
   398→  const treasuryPagePadMap = ['8px 10px', '6px 8px', '4px 6px']
   399→  const sl = globalSizeLevel
   400→
   401→  // Split rooms across pages: max 6 rooms per page (3 rows × 2 cols)
   402→  const MAX_ROOMS_PER_PAGE = 6
   403→  const pages: string[] = []
   404→
   405→  const totalFullPages = Math.floor(roomCells.length / MAX_ROOMS_PER_PAGE)
   406→  const hasRemainder = roomCells.length % MAX_ROOMS_PER_PAGE > 0
   407→
   408→  // Full room pages - ALL use global adaptive sizeLevel
   409→  for (let p = 0; p < totalFullPages; p++) {
   410→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   411→    const isFullPage = pageRooms.length === MAX_ROOMS_PER_PAGE
   412→    // Full pages (e.g. 6 rooms): no min-height/centering to prevent last table cut-off
   413→    // Partial pages: keep min-height for clean layout
   414→    const containerStyle = isFullPage
   415→      ? 'width:780px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   416→      : 'width:780px;min-height:1120px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   417→    const contentStyle = isFullPage
   418→      ? 'display:flex;flex-direction:column;align-items:center;'
   419→      : 'flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;'
   420→    pages.push(
   421→      '<div style="' + containerStyle + '" dir="rtl">' +
   422→      buildHeader(sl) +
   423→      '<div style="' + contentStyle + '">' +
   424→      buildRoomsGrid(pageRooms, sl) +
   425→      '</div>' +
   426→      '</div>'
   427→    )
   428→  }
   429→
   430→  // Overflow rooms for treasury page
   431→  const overflowRoomCount = hasRemainder ? roomCells.length % MAX_ROOMS_PER_PAGE : 0
   432→  const overflowCells = overflowRoomCount > 0 ? roomCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE) : []
   433→
   434→  // Treasury page - uses same global sizeLevel
   435→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses, sl)
   436→  const overflowHtml = overflowCells.length > 0 ? buildRoomsGrid(overflowCells, sl) : ''
   437→
   438→  // Split overflow rooms and treasury onto pages
   439→  if (overflowHtml) {
   440→    if (overflowRoomCount <= 3 && totalFullPages > 0) {
   441→      // غرف زائدة قليلة مع وجود صفحات غرف كاملة → ندمجها مع الخزينة في صفحة واحدة
   442→      pages.push(
   443→        '<div style="width:780px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   444→        buildHeader(sl) +
   445→        overflowHtml +
   446→        '<div style="margin-top:16px;">' +
   447→        treasuryContent +
   448→        '</div>' +
   449→        '</div>'
   450→      )
   451→    } else {
   452→      // إما غرف زائدة كثيرة أو لا توجد صفحات غرف كاملة (مثل المنصوره 3 غرف)
   453→      // → صفحة منفصلة للغرف + صفحة منفصلة للخزينة
   454→      pages.push(
   455→        '<div style="width:780px;min-height:1120px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;" dir="rtl">' +
   456→        buildHeader(sl) +
   457→        '<div style="flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;">' +
   458→        overflowHtml +
   459→        '</div>' +
   460→        '</div>'
   461→      )
   462→      pages.push(
   463→        '<div style="width:780px;min-height:1120px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   464→        buildHeader(sl) +
   465→        treasuryContent +
   466→        '</div>'
   467→      )
   468→    }
   469→  } else {
   470→    // No overflow rooms - treasury is the only content on this page
   471→    pages.push(
   472→      '<div style="width:780px;min-height:1120px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   473→      buildHeader(sl) +
   474→      treasuryContent +
   475→      '</div>'
   476→    )
   477→  }
   478→
   479→  return pages
   480→}
