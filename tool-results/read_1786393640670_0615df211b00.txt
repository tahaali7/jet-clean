   121→function getRoomsForBranch(branchName: string) { return BRANCH_ROOMS[branchName] || ROOMS }
   122→function getPricesForRoom(room: string) { return ROOM_PRICES[room] || ALL_PRICES }
   123→function isExtraEnabledForBranch(branchName: string) { return !BRANCH_EXTRA_DISABLED.includes(branchName) }
   124→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   125→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   126→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   127→  return Math.floor(totalAmount / 2) - deduction
   128→}
   129→function getNetFormulaText(branchName: string, roomName: string) {
   130→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return 'الإجمالي ÷ 2'
   131→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   132→  if (deduction === 0) return 'الإجمالي ÷ 2'
   133→  return `الإجمالي ÷ 2 - ${deduction}`
   134→}
   135→function todayISO() {
   136→  const d = new Date()
   137→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   138→}
   139→function formatDateShort(dateStr: string) {
   140→  if (!dateStr) return ''
   141→  const [y, m, d] = dateStr.split('-')
   142→  return `${d}/${m}/${y}`
   143→}

   315→  // Group entries by room

   323→  let grandTotalAmount = 0

   329→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   330→  const orderedRooms = [...regularRooms]
   331→  if (hasMachine) orderedRooms.push('مكينة الغسيل')

   344→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)

   346→        grandTotalAmount += roomTotal
   347→        grandTotalCars += roomCars
   348→        grandTotalNet += getNetAmount(roomTotal, branchName, room)

   350→      } else {
   351→        cells.push(buildEmptyRoomTableHTML(room, sizeLevel))
   352→      }
   353→    })

   392→  const treasuryPagePadMap = ['8px 10px', '6px 8px', '4px 6px']
   393→  const sl = globalSizeLevel

   395→  // Split rooms across pages: max 6 rooms per page (3 rows × 2 cols)
   396→  const MAX_ROOMS_PER_PAGE = 6
   397→  const pages: string[] = []

   402→  // Full room pages - ALL use global adaptive sizeLevel
   403→  for (let p = 0; p < totalFullPages; p++) {
   404→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   405→    const isFullPage = pageRooms.length === MAX_ROOMS_PER_PAGE
   406→    // Full pages (e.g. 6 rooms): no min-height/centering to prevent last table cut-off
   407→    // Partial pages: keep min-height for clean layout
   408→    const containerStyle = isFullPage
   409→      ? 'width:780px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   410→      : 'width:780px;min-height:1120px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   411→    const contentStyle = isFullPage
   412→      ? 'display:flex;flex-direction:column;align-items:center;'
   413→      : 'flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;'
   414→    pages.push(
   415→      '<div style="' + containerStyle + '" dir="rtl">' +
   416→      buildHeader(sl) +
   417→      '<div style="' + contentStyle + '">' +
   418→      buildRoomsGrid(pageRooms, sl) +
   419→      '</div>' +
   420→      '</div>'
   421→    )
   422→  }
   423→
   424→  // Overflow rooms for treasury page
   425→  const overflowRoomCount = hasRemainder ? roomCells.length % MAX_ROOMS_PER_PAGE : 0
   426→  const overflowCells = overflowRoomCount > 0 ? roomCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE) : []
   427→
   428→  // Treasury page - uses same global sizeLevel
   429→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses, sl)
   430→  const overflowHtml = overflowCells.length > 0 ? buildRoomsGrid(overflowCells, sl) : ''
   431→  pages.push(
   432→    '<div style="width:780px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   433→    buildHeader(sl) +
   434→    overflowHtml +
   435→    treasuryContent +
   436→    '</div>'
   437→  )
   438→
   439→  return pages
   440→}

   750→
   751→  // Render each branch as a separate block, then pack into pages
   752→  const branchBlocks: { html: string; estimatedHeight: number }[] = []
   753→  branchDatas.forEach(bd => {
   754→    const blockHtml = '<div style="margin-bottom:16px;">' +
   755→      '<h3 style="background:#0e7490;color:#fff;padding:6px 10px;border-radius:6px;font-size:13px;margin-bottom:6px;">' +
   756→      'فرع ' + bd.name + ' — سحوبات: ' + bd.withdrawals + ' د.ل | عجوزات: ' + bd.shortages + ' د.ل | الإجمالي: ' + (bd.withdrawals + bd.shortages) + ' د.ل' +
   757→      '</h3>' +
   758→      '<table style="width:100%;border-collapse:collapse;font-size:11px;">' +
   759→      '<thead><tr style="background:#e2e8f0;">' +
   760→      '<th style="padding:6px;border:1px solid #ddd;">الموظف</th>' +
   761→      '<th style="padding:6px;border:1px solid #ddd;">السحبيات</th>' +
   762→      '<th style="padding:6px;border:1px solid #ddd;">العجوزات</th>' +
   763→      '<th style="padding:6px;border:1px solid #ddd;">الإجمالي</th>' +
   764→      '</tr></thead>' +
   765→      '<tbody>' + bd.empsHtml + '</tbody>' +
   766→      '</table></div>'
   767→    // Estimate: title ~30px + header row ~30px + each emp row ~35px + detail rows ~25px each
   768→    const rowCount = bd.empsHtml.split('<tr').length
   769→    branchBlocks.push({ html: blockHtml, estimatedHeight: 30 + 30 + rowCount * 35 })
   770→  })
   771→
   772→  // Header + summary estimated ~200px, footer ~30px, page usable ~880px (1120 - 200 - 30)
   773→  const headerHeight = 200
   774→  const footerHeight = 30
   775→  const usableHeight = 890
   776→
   777→  let pageNum = 1
   778→  let currentPageContent = headerHtml + summaryHtml
   779→  let currentPageUsed = headerHeight
   780→
   781→  branchBlocks.forEach((block, idx) => {
   782→    if (currentPageUsed + block.estimatedHeight + footerHeight > 1120 && idx > 0) {
   783→      // Current page is full, start a new one
   784→      currentPageContent += footerHtml.replace('__PAGE__', String(pageNum))
   785→      pages.push('<div style="' + pageWithMinHeight + '">' + currentPageContent + '</div>')
   786→      pageNum++
   787→      currentPageContent = headerHtml + block.html
   788→      currentPageUsed = headerHeight + block.estimatedHeight
   789→    } else {
   790→      currentPageContent += block.html
   791→      currentPageUsed += block.estimatedHeight
   792→    }
   793→  })
   794→
   795→  // Last page
   796→  if (currentPageContent) {
   797→    currentPageContent += footerHtml.replace('__PAGE__', String(pageNum))
   798→    const isLastPage = branchBlocks.length <= 2
   799→    pages.push('<div style="' + (isLastPage ? pageWithMinHeight : pageStyle) + '">' + currentPageContent + '</div>')
   800→  }
   801→
   802→  return pages
   803→}
