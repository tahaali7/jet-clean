   832→    const isLastPage = branchBlocks.length <= 2
   833→    pages.push('<div style="' + (isLastPage ? pageWithMinHeight : pageStyle) + '">' + currentPageContent + '</div>')
   834→  }
   835→
   836→  return pages
   837→}
