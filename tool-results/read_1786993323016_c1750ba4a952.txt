   127→function getRoomsForBranch(branchName: string) { return BRANCH_ROOMS[branchName] || ROOMS }
   128→function getPricesForRoom(room: string) { return ROOM_PRICES[room] || ALL_PRICES }
   129→function isExtraEnabledForBranch(branchName: string) { return !BRANCH_EXTRA_DISABLED.includes(branchName) }
   130→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   131→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   132→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   133→  return Math.floor(totalAmount / 2) - deduction
   134→}
   135→function getNetFormulaText(branchName: string, roomName: string) {
   136→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return 'الإجمالي ÷ 2'
   137→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   138→  if (deduction === 0) return 'الإجمالي ÷ 2'
   139→  return `الإجمالي ÷ 2 - ${deduction}`
   140→}
   141→function todayISO() {
   142→  const d = new Date()

   144→}
   145→function formatDateShort(dateStr: string) {
   146→  if (!dateStr) return ''
   147→  const [y, m, d] = dateStr.split('-')

   156→  let roomTotalCars = 0

   159→  const mergedCounts: Record<string, number> = {}
   160→  const mergedCustoms: Record<string, { price: number; count: number }> = {}
   161→
   162→  roomEntries.forEach(entry => {
   163→    prices.forEach(price => {
   164→      const count = (entry.priceCounts && entry.priceCounts[price]) || 0
   165→      mergedCounts[price] = (mergedCounts[price] || 0) + count
   166→    })
   167→    if (entry.customPrices) {
   168→      Object.keys(entry.customPrices).forEach(key => {
   169→        const item = entry.customPrices[key]
   170→        if (mergedCustoms[key]) {
   171→          mergedCustoms[key].count += item.count
   172→        } else {
   173→          mergedCustoms[key] = { price: item.price, count: item.count }
   174→        }
   175→      })
   176→    }
   177→    roomTotalAmount += entry.totalAmount
   178→    roomTotalCars += entry.totalCars
   179→    roomExtraCars += (entry.extraCars || 0)
   180→    roomExtraAmount += (entry.extraAmount || 0)
   181→  })
   182→
   183→  const roomNet = getNetAmount(roomTotalAmount, branchName, room)
   184→  // Auto-adaptive: level 0=normal, 1=compact, 2=ultra-compact
   185→  const sl = sizeLevel || 0
   186→  const padMap = ['10px 10px', '5px 7px', '3px 5px']
   187→  const fsMap = ['font-size:10px;', 'font-size:9px;', 'font-size:8px;']
   188→  const titlePadMap = ['padding:10px 10px;', 'padding:5px 7px;', 'padding:3px 5px;']

   191→  const _pad = padMap[sl]
   192→  const cellPad = 'padding:' + _pad + ';vertical-align:middle;'
   193→  const cellFs = fsMap[sl]
   194→  const titlePad = titlePadMap[sl]
   195→  const titleFs = titleFsMap[sl]
   196→  const countFs = countFsMap[sl]
   197→
   198→  let rowsHtml = ''
   199→  let rowNum = 0
   200→  prices.forEach((price, idx) => {
   201→    const count = mergedCounts[price] || 0
   202→    rowNum++
   203→    const isExtra = EXTRA_PRICES.includes(price) && isExtraEnabledForBranch(branchName)
   204→    const displayPrice = isExtra ? (price - 5) : price
   205→    const rowAmount = displayPrice * count
   206→    rowsHtml += '<tr>' +
   207→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + '">' + rowNum + '</td>' +
   208→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + '">' + displayPrice + ' د.ل</td>' +
   209→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + countFs + 'font-weight:bold;">' + count + '</td>' +
   210→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + '">' + rowAmount + ' د.ل</td>' +
   211→    '</tr>'
   212→  })
   213→
   214→  const customKeys = Object.keys(mergedCustoms)
   215→  customKeys.forEach(key => {
   216→    const item = mergedCustoms[key]
   217→    rowNum++
   218→    rowsHtml += '<tr>' +
   219→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + 'color:#7c3aed;">✦</td>' +
   220→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + 'color:#7c3aed;">' + item.price + ' د.ل</td>' +
   221→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + countFs + 'font-weight:bold;">' + item.count + '</td>' +
   222→      '<td style="' + cellPad + 'border:1px solid #555;text-align:center;' + cellFs + 'font-weight:bold;">' + (item.price * item.count) + ' د.ل</td>' +
   223→    '</tr>'
   224→  })
   225→
   226→  let extraRowHtml = ''
   227→  if (roomExtraCars > 0) {
   228→    extraRowHtml = '<tr style="background:#fffde7;">' +
   229→      '<td colspan="2" style="' + cellPad + 'border:1px solid #555;' + cellFs + 'font-weight:bold;text-align:center;color:#e65100;">⭐ إكسترا</td>' +

   250→      roomAddonRowsHtml += '<tr style="background:#f3e5f5;">' +
   251→        '<td colspan="2" style="' + cellPad + 'border:1px solid #555;' + cellFs + 'font-weight:bold;text-align:center;color:#7b1fa2;">🔧 الصيانة</td>' +

   259→    '<colgroup><col style="width:10%;"/><col style="width:28%;"/><col style="width:24%;"/><col style="width:38%;"/></colgroup>' +
   260→    '<tr><td colspan="4" style="' + titlePad + 'text-align:center;' + titleFs + 'font-weight:bold;background:#bdbdbd;color:#222;">' + room + '</td></tr>' +
   261→    '<tr style="background:#e0e0e0;">' +

   345→function buildCarReportHTML(selectedDate: string, branchId: string, branchName: string, entries: CarEntry[], savedWorkerExpenses?: Record<string, { cleanliness?: number; treasury?: Record<string, { income: number; expense: number }> }>): string[] {
   346→  // Group entries by room
   347→  const roomMap: Record<string, CarEntry[]> = {}
   348→  entries.forEach(e => {
   349→    if (!roomMap[e.room]) roomMap[e.room] = []
   350→    roomMap[e.room].push(e)
   351→  })
   352→
   353→  const branchRooms = getRoomsForBranch(branchName)
   354→  let grandTotalAmount = 0
   355→  let grandTotalCars = 0
   356→  let grandTotalNet = 0
   357→
   358→  // Order rooms: regular rooms first, machine last
   359→  const regularRooms = branchRooms.filter(r => r !== 'مكينة الغسيل')
   360→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   361→  const orderedRooms = [...regularRooms]
   362→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   363→
   364→  // Auto-adaptive: calculate GLOBAL size level based on total rooms
   365→  // 0 = normal (<=5 rooms), 1 = compact (never used currently), 2 = ultra-compact (6+)
   366→  const totalRoomCount = orderedRooms.length
   367→  const globalSizeLevel = totalRoomCount <= 5 ? 0 : 2
   368→
   369→  // Build room data with global adaptive sizing
   370→  const buildRoomCells = (sizeLevel: number) => {
   371→    const cells: string[] = []
   372→    orderedRooms.forEach(room => {
   373→      const roomEntries = roomMap[room] || []
   374→      if (roomEntries.length > 0) {
   375→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   376→        const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
   377→        grandTotalAmount += roomTotal
   378→        grandTotalCars += roomCars
   379→        grandTotalNet += getNetAmount(roomTotal, branchName, room)
   380→        cells.push(buildRoomTableHTML(room, roomEntries, branchName, sizeLevel))
   381→      } else {
   382→        cells.push(buildEmptyRoomTableHTML(room, sizeLevel))
   383→      }
   384→    })
   385→    return cells
   386→  }
   387→
   388→  // Build ALL room cells with global sizeLevel
   389→  const roomCells = buildRoomCells(globalSizeLevel)
   390→
   391→  // Helper: build rooms grid HTML from array of room cells (adaptive)
   392→  const buildRoomsGrid = (cells: string[], sizeLevel?: number) => {
   393→    let html = ''
   394→    const sl = sizeLevel || 0

   399→    for (let i = 0; i < cells.length; i += 2) {

   401→        '<div style="flex:1;min-width:0;">' + cells[i] + '</div>' +

   411→    const fsMap = ['22px', '18px', '15px']
   412→    const dateFsMap = ['14px', '12px', '10px']
   413→    const mbMap = ['14px', '10px', '6px']
   414→    const pbMap = ['10px', '6px', '4px']

   416→      '<h1 style="font-size:' + fsMap[sl] + ';font-weight:bold;margin:0 0 6px 0;">مغسلة <span style="margin:0 0 0 15px;">' + branchName + '</span></h1>' +
   417→      '<p style="font-size:' + dateFsMap[sl] + ';margin:0;color:#000000;">التاريخ: ' + formatDateShort(selectedDate) + '</p>' +
   418→      '</div>'
   419→  }
   420→

   422→  const roomPagePadMap = ['12px 10px', '10px 8px', '8px 6px']
   423→  const treasuryPagePadMap = ['8px 10px', '6px 8px', '4px 6px']
   424→  const sl = globalSizeLevel
   425→
   426→  // Split rooms across pages: max 6 rooms per page (3 rows × 2 cols)
   427→  const MAX_ROOMS_PER_PAGE = 6
   428→  const pages: string[] = []
   429→
   430→  const totalFullPages = Math.floor(roomCells.length / MAX_ROOMS_PER_PAGE)
   431→  const hasRemainder = roomCells.length % MAX_ROOMS_PER_PAGE > 0
   432→
   433→  // Full room pages - ALL use global adaptive sizeLevel
   434→  for (let p = 0; p < totalFullPages; p++) {
   435→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   436→    const isFullPage = pageRooms.length === MAX_ROOMS_PER_PAGE

   442→    const contentStyle = isFullPage

   444→      : 'flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;'

   450→      '</div>' +
   451→      '</div>'

   453→  }
   454→
   455→  // Overflow rooms for treasury page
   456→  const overflowRoomCount = hasRemainder ? roomCells.length % MAX_ROOMS_PER_PAGE : 0
   457→  const overflowCells = overflowRoomCount > 0 ? roomCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE) : []

   466→      // غرف زائدة قليلة مع وجود صفحات غرف كاملة → ندمجها مع الخزينة في صفحة واحدة

   469→        buildHeader(sl) +
   470→        overflowHtml +

   472→        treasuryContent +
   473→        '</div>' +
   474→        '</div>'
   475→      )

   477→      // إما غرف زائدة كثيرة أو لا توجد صفحات غرف كاملة (مثل المنصوره 3 غرف)
   478→      // → صفحة منفصلة للغرف + صفحة منفصلة للخزينة
   479→      pages.push(
   480→        '<div style="width:780px;min-height:1120px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;" dir="rtl">' +
   481→        buildHeader(sl) +
   482→        '<div style="flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;">' +
   483→        overflowHtml +
   484→        '</div>' +
   485→        '</div>'
   486→      )
   487→      pages.push(
   488→        '<div style="width:780px;min-height:1120px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   489→        buildHeader(sl) +
   490→        treasuryContent +
   491→        '</div>'
   492→      )
   493→    }
   494→  } else {
   495→    // No overflow rooms - treasury is the only content on this page
   496→    pages.push(
   497→      '<div style="width:780px;min-height:1120px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   498→      buildHeader(sl) +
   499→      treasuryContent +
   500→      '</div>'
   501→    )
   502→  }
   503→
   504→  return pages
   505→}

   638→      '<td style="' + tValueStyle + 'color:#c62828;">' + (tExpense > 0 ? tExpense : '') + '</td>' +
   639→      '<td style="' + tValueStyle + 'color:' + balColor + ';">' + tRunningBalance + '</td>' +
   640→    '</tr>'
   641→  })
   642→

   644→    '<table style="width:100%;border-collapse:collapse;font-family:Cairo,sans-serif;">' +
   645→    '<tr><td colspan="4" style="' + wTitlePad + 'text-align:center;' + wTitleFs + 'font-weight:bold;border:1px solid #555;">الخزينة</td></tr>' +
   646→    treasuryRowsHtml +
   647→    '</table>' +
   648→    '</div>'
   649→
   650→  return '<div style="' + sepMtMap[sl] + 'border-top:2px solid #000;"><div style="display:flex;' + sepGapMap[sl] + '">' + workerExpensesHtml + treasuryHtml + '</div></div>'
