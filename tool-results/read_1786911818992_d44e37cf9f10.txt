   703→        async function exportClosingPDF() {
   704→            const selectedDate = document.getElementById('datePicker').value;
   705→            const periodLabel = `إغلاق يومي: ${formatDateShort(selectedDate)}`;
   706→            const matchRecord = r => r.date === selectedDate;
   707→
   708→            try {
   709→                const reportArea = document.getElementById('pdfReportArea');
   710→                reportArea.innerHTML = buildReportHTML(periodLabel, matchRecord);
   711→                await new Promise(resolve => setTimeout(resolve, 100));
   712→
   713→                const canvas = await html2canvas(reportArea, { scale: 2, backgroundColor: '#ffffff', useCORS: true });
   714→                const imgData = canvas.toDataURL('image/png');
   715→
   716→                const { jsPDF } = window.jspdf;
   717→                const pdf = new jsPDF('p', 'mm', 'a4');
   718→                const pageWidth = pdf.internal.pageSize.getWidth();
   719→                const pageHeight = pdf.internal.pageSize.getHeight();
   720→                const imgWidth = pageWidth;
   721→                const imgHeight = (canvas.height * imgWidth) / canvas.width;
   722→
   723→                let heightLeft = imgHeight;
   724→                let position = 0;
   725→                pdf.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
   726→                heightLeft -= pageHeight;
   727→                while (heightLeft > 0) {
   728→                    position = heightLeft - imgHeight;
   729→                    pdf.addPage();
   730→                    pdf.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
   731→                    heightLeft -= pageHeight;
   732→                }
   733→
   734→                pdf.save(`إغلاق_يومي_${selectedDate}.pdf`);
   735→                reportArea.innerHTML = '';
   736→            } catch (err) {
   737→                console.error(err);
   738→                alert('حدث خطأ أثناء إنشاء ملف PDF. الرجاء المحاولة مرة أخرى.');
   739→            }
   740→        }

   761→        async function exportPDF() {
   762→            const type = document.getElementById('exportRangeType').value;
   763→
   764→            let periodLabel, matchRecord, fileSuffix;
   765→
   766→            if (type === 'month') {
   767→                const monthValue = document.getElementById('exportMonth').value;
   768→                if (!monthValue) return alert('الرجاء اختيار الشهر');
   769→                periodLabel = `الشهر: ${formatMonthLabel(monthValue)}`;
   770→                matchRecord = r => r.month === monthValue;
   771→                fileSuffix = formatMonthLabel(monthValue).replace(/\s+/g, '_');
   772→            } else if (type === 'day') {
   773→                const dayValue = document.getElementById('exportDay').value;
   774→                if (!dayValue) return alert('الرجاء اختيار اليوم');
   775→                periodLabel = `اليوم: ${formatDateShort(dayValue)}`;
   776→                matchRecord = r => r.date === dayValue;
   777→                fileSuffix = dayValue;
   778→            } else {
   779→                const from = document.getElementById('exportFrom').value;
   780→                const to = document.getElementById('exportTo').value;
   781→                if (!from || !to) return alert('الرجاء تحديد بداية ونهاية الفترة');
   782→                if (from > to) return alert('تاريخ البداية يجب أن يكون قبل تاريخ النهاية');
   783→                periodLabel = `من ${formatDateShort(from)} إلى ${formatDateShort(to)}`;
   784→                matchRecord = r => r.date >= from && r.date <= to;
   785→                fileSuffix = `${from}_${to}`;
