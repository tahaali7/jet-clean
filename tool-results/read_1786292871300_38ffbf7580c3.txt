   319→   319→    if (!roomMap[e.room]) roomMap[e.room] = []

   322→   322→

   324→   324→  let grandTotalAmount = 0

   331→   331→  const orderedRooms = [...regularRooms]
   332→   332→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   333→   333→
   334→   334→  // Auto-adaptive: calculate GLOBAL size level based on total rooms

   336→   336→  const totalRoomCount = orderedRooms.length
   337→   337→  const globalSizeLevel = totalRoomCount <= 5 ? 0 : 2
   338→   338→
   339→   339→  // Build room data with global adaptive sizing
   340→   340→  const buildRoomCells = (sizeLevel: number) => {
   341→   341→    const cells: string[] = []
   342→   342→    orderedRooms.forEach(room => {
   343→   343→      const roomEntries = roomMap[room] || []
   344→   344→      if (roomEntries.length > 0) {
   345→   345→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   346→   346→        const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
