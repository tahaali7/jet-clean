   119→function getRoomsForBranch(branchName: string) { return BRANCH_ROOMS[branchName] || ROOMS }
   120→function getPricesForRoom(room: string) { return ROOM_PRICES[room] || ALL_PRICES }
   121→function getNetAmount(totalAmount: number, branchName: string, roomName: string) {
   122→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return Math.floor(totalAmount / 2)
   123→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   124→  return Math.floor(totalAmount / 2) - deduction
   125→}
   126→function getNetFormulaText(branchName: string, roomName: string) {
   127→  if (roomName === 'مكينة الغسيل' && MACHINE_NO_DEDUCTION_BRANCHES.includes(branchName)) return 'الإجمالي ÷ 2'
   128→  const deduction = BRANCH_NET_DEDUCTION[branchName] !== undefined ? BRANCH_NET_DEDUCTION[branchName] : 0
   129→  if (deduction === 0) return 'الإجمالي ÷ 2'
   130→  return `الإجمالي ÷ 2 - ${deduction}`
   131→}
   132→function todayISO() {
   133→  const d = new Date()
   134→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   135→}
   136→function formatDateShort(dateStr: string) {
   137→  if (!dateStr) return ''
   138→  const [y, m, d] = dateStr.split('-')
   139→  return `${d}/${m}/${y}`
   140→}

   154→    prices.forEach(price => {
   155→      const count = (entry.priceCounts && entry.priceCounts[price]) || 0

   313→  // Group entries by room
   314→  const roomMap: Record<string, CarEntry[]> = {}
   315→  entries.forEach(e => {
   316→    if (!roomMap[e.room]) roomMap[e.room] = []
   317→    roomMap[e.room].push(e)

   319→
   320→  const branchRooms = getRoomsForBranch(branchName)
   321→  let grandTotalAmount = 0
   322→  let grandTotalCars = 0
   323→  let grandTotalNet = 0
   324→
   325→  // Order rooms: regular rooms first, machine last

   327→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   328→  const orderedRooms = [...regularRooms]
   329→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   330→
   331→  // Auto-adaptive: calculate GLOBAL size level based on total rooms

   334→  const globalSizeLevel = totalRoomCount <= 5 ? 0 : 2

   337→  const buildRoomCells = (sizeLevel: number) => {

   339→    orderedRooms.forEach(room => {
   340→      const roomEntries = roomMap[room] || []
   341→      if (roomEntries.length > 0) {
   342→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   343→        const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
   344→        grandTotalAmount += roomTotal
   345→        grandTotalCars += roomCars
   346→        grandTotalNet += getNetAmount(roomTotal, branchName, room)
   347→        cells.push(buildRoomTableHTML(room, roomEntries, branchName, sizeLevel))
   348→      } else {
   349→        cells.push(buildEmptyRoomTableHTML(room, sizeLevel))
   350→      }
   351→    })
   352→    return cells
   353→  }
   354→
   355→  // Build ALL room cells with global sizeLevel
   356→  const roomCells = buildRoomCells(globalSizeLevel)

   358→  // Helper: build rooms grid HTML from array of room cells (adaptive)

   390→  const treasuryPagePadMap = ['8px 10px', '6px 8px', '4px 6px']
   391→  const sl = globalSizeLevel
   392→
   393→  // Split rooms across pages: max 6 rooms per page (3 rows × 2 cols)
   394→  const MAX_ROOMS_PER_PAGE = 6
   395→  const pages: string[] = []
   396→
   397→  const totalFullPages = Math.floor(roomCells.length / MAX_ROOMS_PER_PAGE)
   398→  const hasRemainder = roomCells.length % MAX_ROOMS_PER_PAGE > 0
   399→
   400→  // Full room pages - ALL use global adaptive sizeLevel
   401→  for (let p = 0; p < totalFullPages; p++) {
   402→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   403→    const isFullPage = pageRooms.length === MAX_ROOMS_PER_PAGE
   404→    // Full pages (e.g. 6 rooms): no min-height/centering to prevent last table cut-off
   405→    // Partial pages: keep min-height for clean layout
   406→    const containerStyle = isFullPage
   407→      ? 'width:780px;background:#fff;color:#000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   408→      : 'width:780px;min-height:1120px;background:#fff;color:#000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   409→    const contentStyle = isFullPage
   410→      ? 'display:flex;flex-direction:column;align-items:center;'
   411→      : 'flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;'
   412→    pages.push(
   413→      '<div style="' + containerStyle + '" dir="rtl">' +
   414→      buildHeader(sl) +
   415→      '<div style="' + contentStyle + '">' +
   416→      buildRoomsGrid(pageRooms, sl) +
   417→      '</div>' +
   418→      '</div>'
   419→    )
   420→  }
   421→
   422→  // Overflow rooms for treasury page
   423→  const overflowRoomCount = hasRemainder ? roomCells.length % MAX_ROOMS_PER_PAGE : 0
   424→  const overflowCells = overflowRoomCount > 0 ? roomCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE) : []
   425→
   426→  // Treasury page - uses same global sizeLevel
   427→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses, sl)
   428→  const overflowHtml = overflowCells.length > 0 ? buildRoomsGrid(overflowCells, sl) : ''
   429→  pages.push(
   430→    '<div style="width:780px;background:#fff;color:#000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   431→    buildHeader(sl) +
   432→    overflowHtml +
   433→    treasuryContent +
   434→    '</div>'
   435→  )
   436→
   437→  return pages
   438→}

   757→  branchDatas.forEach(bd => {

   788→      currentPageContent += footerHtml.replace('__PAGE__', String(pageNum))
   789→      pages.push('<div style="' + pageWithMinHeight + '">' + currentPageContent + '</div>')
   790→      pageNum++
   791→      currentPageContent = headerHtml + block.html
   792→      currentPageUsed = headerHeight + block.estimatedHeight
   793→    } else {
   794→      currentPageContent += block.html
   795→      currentPageUsed += block.estimatedHeight

   802→    const isLastPage = branchBlocks.length <= 2

   804→  }
   805→
   806→  return pages
