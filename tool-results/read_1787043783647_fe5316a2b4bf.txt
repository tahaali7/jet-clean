   353→function buildCarReportHTML(selectedDate: string, branchId: string, branchName: string, entries: CarEntry[], savedWorkerExpenses?: Record<string, { cleanliness?: number; treasury?: Record<string, { income: number; expense: number }> }>): string[] {
   354→  // Group entries by room
   355→  const roomMap: Record<string, CarEntry[]> = {}
   356→  entries.forEach(e => {
   357→    if (!roomMap[e.room]) roomMap[e.room] = []
   358→    roomMap[e.room].push(e)
   359→  })
   360→
   361→  const branchRooms = getRoomsForBranch(branchName)
   362→  let grandTotalAmount = 0
   363→  let grandTotalCars = 0
   364→  let grandTotalNet = 0
   365→
   366→  // Order rooms: regular rooms first, machine last
   367→  const regularRooms = branchRooms.filter(r => r !== 'مكينة الغسيل')
   368→  const hasMachine = branchRooms.includes('مكينة الغسيل')
   369→  const orderedRooms = [...regularRooms]
   370→  if (hasMachine) orderedRooms.push('مكينة الغسيل')
   371→
   372→  // Auto-adaptive: calculate GLOBAL size level based on total rooms
   373→  // 0 = normal (<=5 rooms), 1 = compact (never used currently), 2 = ultra-compact (6+)
   374→  const totalRoomCount = orderedRooms.length
   375→  const globalSizeLevel = totalRoomCount <= 5 ? 0 : 2
   376→
   377→  // Build room data with global adaptive sizing
   378→  const buildRoomCells = (sizeLevel: number) => {
   379→    const cells: string[] = []
   380→    orderedRooms.forEach((room, idx) => {
   381→      const roomEntries = roomMap[room] || []
   382→      if (roomEntries.length > 0) {
   383→        const roomTotal = roomEntries.reduce((s, e) => s + e.totalAmount, 0)
   384→        const roomCars = roomEntries.reduce((s, e) => s + e.totalCars, 0)
   385→        grandTotalAmount += roomTotal
   386→        grandTotalCars += roomCars
   387→        grandTotalNet += getNetAmount(roomTotal, branchName, room)
   388→        cells.push(buildRoomTableHTML(room, roomEntries, branchName, sizeLevel, idx))
   389→      } else {
   390→        cells.push(buildEmptyRoomTableHTML(room, sizeLevel, idx))
   391→      }
   392→    })
   393→    return cells
   394→  }
   395→
   396→  // Build ALL room cells with global sizeLevel
   397→  const roomCells = buildRoomCells(globalSizeLevel)
   398→
   399→  // Helper: build rooms grid HTML from array of room cells (adaptive)
   400→  const buildRoomsGrid = (cells: string[], sizeLevel?: number) => {
   401→    let html = ''
   402→    const sl = sizeLevel || 0
   403→    const gapMap = ['10px', '7px', '5px']
   404→    const mbMap = ['10px', '7px', '5px']

   406→    const mb = mbMap[sl]
   407→    for (let i = 0; i < cells.length; i += 2) {
   408→      html += '<div style="display:flex;gap:' + gap + ';margin-bottom:' + mb + ';">' +
   409→        '<div style="flex:1;min-width:0;">' + cells[i] + '</div>' +
   410→        (cells[i + 1] ? '<div style="flex:1;min-width:0;">' + cells[i + 1] + '</div>' : '') +
   411→        '</div>'
   412→    }
   413→    return html
   414→  }
   415→
   416→  // Helper: build page header (adaptive)
   417→  const buildHeader = (sizeLevel?: number) => {
   418→    const sl = sizeLevel || 0
   419→    const fsMap = ['22px', '18px', '15px']
   420→    const dateFsMap = ['14px', '12px', '10px']
   421→    const mbMap = ['14px', '10px', '6px']
   422→    const padMap2 = ['10px 12px', '7px 10px', '5px 8px']
   423→    return '<div style="text-align:center;margin-bottom:' + mbMap[sl] + ';background:linear-gradient(to left, #1F497D, #31869A);color:#FFFFFF;padding:' + padMap2[sl] + ';border-radius:4px;">' +
   424→      '<h1 style="font-size:' + fsMap[sl] + ';font-weight:bold;margin:0 0 4px 0;color:#FFFFFF;">مغسلة <span style="margin:0 0 0 15px;">' + branchName + '</span></h1>' +
   425→      '<p style="font-size:' + dateFsMap[sl] + ';margin:0;color:#E8EEF7;">التاريخ: ' + formatDateShort(selectedDate) + '</p>' +
   426→      '</div>'
   427→  }
   428→
   429→  // Adaptive page container padding
   430→  const roomPagePadMap = ['12px 10px', '10px 8px', '8px 6px']
   431→  const treasuryPagePadMap = ['8px 10px', '6px 8px', '4px 6px']
   432→  const sl = globalSizeLevel
   433→
   434→  // Split rooms across pages: max 6 rooms per page (3 rows × 2 cols)
   435→  const MAX_ROOMS_PER_PAGE = 6
   436→  const pages: string[] = []
   437→
   438→  const totalFullPages = Math.floor(roomCells.length / MAX_ROOMS_PER_PAGE)
   439→  const hasRemainder = roomCells.length % MAX_ROOMS_PER_PAGE > 0
   440→
   441→  // Full room pages - ALL use global adaptive sizeLevel
   442→  for (let p = 0; p < totalFullPages; p++) {
   443→    const pageRooms = roomCells.slice(p * MAX_ROOMS_PER_PAGE, (p + 1) * MAX_ROOMS_PER_PAGE)
   444→    const isFullPage = pageRooms.length === MAX_ROOMS_PER_PAGE
   445→    // Full pages (e.g. 6 rooms): no min-height/centering to prevent last table cut-off
   446→    // Partial pages: keep min-height for clean layout
   447→    const containerStyle = isFullPage
   448→      ? 'width:780px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   449→      : 'width:780px;min-height:1120px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;'
   450→    const contentStyle = isFullPage
   451→      ? 'display:flex;flex-direction:column;align-items:center;'
   452→      : 'flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;'
   453→    pages.push(
   454→      '<div style="' + containerStyle + '" dir="rtl">' +
   455→      buildHeader(sl) +
   456→      '<div style="' + contentStyle + '">' +
   457→      buildRoomsGrid(pageRooms, sl) +
   458→      '</div>' +
   459→      '</div>'
   460→    )
   461→  }
   462→
   463→  // Overflow rooms for treasury page
   464→  const overflowRoomCount = hasRemainder ? roomCells.length % MAX_ROOMS_PER_PAGE : 0
   465→  const overflowCells = overflowRoomCount > 0 ? roomCells.slice(totalFullPages * MAX_ROOMS_PER_PAGE) : []
   466→
   467→  // Treasury page - uses same global sizeLevel
   468→  const treasuryContent = buildWorkerExpensesAndTreasury(branchName, selectedDate, orderedRooms, entries, grandTotalNet, savedWorkerExpenses, sl)
   469→  const overflowHtml = overflowCells.length > 0 ? buildRoomsGrid(overflowCells, sl) : ''
   470→
   471→  // Split overflow rooms and treasury onto pages
   472→  if (overflowHtml) {
   473→    if (overflowRoomCount <= 3 && totalFullPages > 0) {
   474→      // غرف زائدة قليلة مع وجود صفحات غرف كاملة → ندمجها مع الخزينة في صفحة واحدة
   475→      pages.push(
   476→        '<div style="width:780px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   477→        buildHeader(sl) +
   478→        overflowHtml +
   479→        '<div style="margin-top:16px;">' +
   480→        treasuryContent +
   481→        '</div>' +
   482→        '</div>'
   483→      )
   484→    } else {
   485→      // إما غرف زائدة كثيرة أو لا توجد صفحات غرف كاملة (مثل المنصوره 3 غرف)
   486→      // → صفحة منفصلة للغرف + صفحة منفصلة للخزينة
   487→      pages.push(
   488→        '<div style="width:780px;min-height:1120px;color:#000000;padding:' + roomPagePadMap[sl] + ';font-family:Cairo,sans-serif;display:flex;flex-direction:column;box-sizing:border-box;" dir="rtl">' +
   489→        buildHeader(sl) +
   490→        '<div style="flex:1;display:flex;flex-direction:column;justify-content:center;align-items:center;">' +
   491→        overflowHtml +
   492→        '</div>' +
   493→        '</div>'
   494→      )
   495→      pages.push(
   496→        '<div style="width:780px;min-height:1120px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   497→        buildHeader(sl) +
   498→        treasuryContent +
   499→        '</div>'
   500→      )
   501→    }
   502→  } else {
   503→    // No overflow rooms - treasury is the only content on this page
   504→    pages.push(
   505→      '<div style="width:780px;min-height:1120px;color:#000000;padding:' + treasuryPagePadMap[sl] + ';font-family:Cairo,sans-serif;" dir="rtl">' +
   506→      buildHeader(sl) +
   507→      treasuryContent +
   508→      '</div>'
   509→    )
   510→  }
   511→
   512→  return pages
   513→}
